144 /**
145 * Checks if the solution obtained at this step after the nonlinear solve is
146 * acceptable or not.
147 *
148 * If adaptive time stepping is disabled or if the simulation is steady-state,
149 * this function always returns true.
150 *
151 * If adaptive time stepping is enabled, this compares the estimated
152 * truncation error to the specified target error for each variable. If the
153 * estimated error is too high, the step is rejected and we try again with a
154 * smaller time step. The passed @p present_solution is set to first vector in
155 * @p previous_solutions, and the current time, time step and step counter are
156 * rolled back to their previous values. If the estimated error is low enough,
157 * the step is accepted and the next time step is set based on the most
158 * critical truncation error estimate.
159 *
160 * Since truncation error is not computed during the starting steps, these
161 * steps are always accepted and the time step is not modified.
162 *
163 * If the time step is rejected too many times in a row, the simulation stops.
164 *
165 * FIXME: Maybe set the maximum number of rejections as a user parameter.
166 * FIXME: Also allow reducing the time step if the Newton method did not
167 * converge.

381template <typename VectorType>
382double TimeHandler::compute_time_derivative(
383 const types::global_dof_index index,
384 const VectorType &present_solution,
385 const std::vector<VectorType> &previous_solutions) const
386{
387 if (scheme == Parameters::TimeIntegration::Scheme::stationary)
388 return 0.;
389 if (scheme == Parameters::TimeIntegration::Scheme::BDF1 ||
390 scheme == Parameters::TimeIntegration::Scheme::BDF2)
391 {
392 double value_dot = bdf_coefficients[0] * present_solution[index];
393 for (unsigned int i = 1; i < bdf_coefficients.size(); ++i)
394 value_dot += bdf_coefficients[i] * previous_solutions[i - 1][index];
395 return value_dot;
396 }
397 DEAL_II_ASSERT_UNREACHABLE();
398}
