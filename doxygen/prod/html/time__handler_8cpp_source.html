37 , max_adaptive_mobility_number(0.)

39 if (!steady_scheme)
40 {
41 // Set starting and final times for each time subinterval (default is 1).
42 const double interval_size =
43 (time_parameters.t_end - time_parameters.t_initial) / n_time_intervals;
44 double t0 = time_parameters.t_initial;
45 for (unsigned int i = 0; i < n_time_intervals; ++i)
46 {
47 initial_times[i] = t0;
48 final_times[i] = t0 + interval_size;
49 t0 += interval_size;
50 }
51
52 // If adaptive time step is disabled (using constant time step), the number
53 // of time subintervals should be compatible with the prescribed time step,
54 // i.e., it should be an integer multiple of dt.
55 if (!with_adaptive_timestep)
56 {
57 const double ratio = interval_size / initial_dt;
58 const double nearest = std::round(ratio);
59 const double diff = std::abs(ratio - nearest);
60 AssertThrow(
61 diff < 1e-10,
62 ExcMessage(
63 "The prescribed constant time step is not compatible with the "
64 "prescribed number of time subintervals. When using a constant time "
65 "step, the length of a subinterval should be an integer multiple of "
66 "the time step. In this case, one time subinterval is " +
67 std::to_string(ratio) + " time steps long."));
68 }
69 }
70
71 switch (scheme)
72 {
73 case STAT:
74 bdf_order = 0;
75 n_previous_solutions = 0;
76 break;
77 case BDF1:
78 bdf_order = 1;
79 n_previous_solutions = 1;
80 break;
81 case BDF2:
82 {
83 bdf_order = 2;
84 n_previous_solutions = 2;
85
86 // Set the initial time step if using BDF1 as starting method
87 if (time_parameters.bdfstart ==
88 Parameters::TimeIntegration::BDFStart::BDF1)
89 {
90 current_dt = initial_dt * time_parameters.bdf_starting_step_ratio;
91 }
92 break;
93 }
94 }
95
96 simulation_times.resize(n_previous_solutions + 1, initial_time);
97 all_simulation_times.push_back(initial_time);
98 time_steps.resize(n_previous_solutions + 1, initial_dt);
99 bdf_coefficients.resize(n_previous_solutions + 1, 0.);
100
101 if (!is_steady() && with_adaptive_timestep)
102 error_estimator =
103 std::make_shared<BDFErrorEstimator>(time_parameters, *this);
104}

185void TimeHandler::set_bdf_coefficients(
186 const bool force_scheme,
187 const Parameters::TimeIntegration::Scheme forced_scheme)
188{
189 const Parameters::TimeIntegration::Scheme used_scheme =
190 force_scheme ? forced_scheme : scheme;
191 switch (used_scheme)
192 {
193 case STAT:
194 bdf_coefficients[0] = 0.;
195 break;
196 case BDF1:
197 {
198 const double dt = time_steps[0];
199 bdf_coefficients[0] = 1. / dt;
200 bdf_coefficients[1] = -1. / dt;
201 break;
202 }
203 case BDF2:
204 {
205 const double dt = time_steps[0];
206 const double prev_dt = time_steps[1];
207 bdf_coefficients[0] = 1.0 / dt + 1.0 / (dt + prev_dt);
208 bdf_coefficients[1] = -1.0 / dt - 1.0 / (prev_dt);
209 bdf_coefficients[2] = dt / prev_dt * 1. / (dt + prev_dt);
210 break;
211 }
212 }

215void TimeHandler::advance(const ConditionalOStream &pcout)
216{
217 current_time_iteration++;
218 current_time_iteration_in_interval++;
219
220 if (scheme == STAT)
221 return;
222
223 // Rotate the times and time steps
224 if (!rolledback_step)
225 for (unsigned int i = n_previous_solutions; i >= 1; --i)
226 {
227 simulation_times[i] = simulation_times[i - 1];
228 time_steps[i] = time_steps[i - 1];
229 }
230
231 // Force another BDF scheme to set the coefficients if this is a starting step
232 bool force_scheme = false;
233 if (scheme == BDF2 &&
234 time_parameters.bdfstart == Parameters::TimeIntegration::BDFStart::BDF1)
235 if (is_starting_step())
236 force_scheme = true;
237
238 // Update the tables and BDF coefficients
239 current_time += current_dt;
240 simulation_times[0] = current_time;
241 time_steps[0] = current_dt;
242 set_bdf_coefficients(force_scheme);
243
244 if (time_parameters.verbosity == Parameters::Verbosity::verbose)
245 {
246 pcout << std::endl
247 << "Time step " << current_time_iteration
248 << " - Advancing to t = " << current_time;
249 if (with_adaptive_timestep &&
250 time_parameters.adaptation.verbosity == Parameters::Verbosity::verbose)
251 pcout << " with time step " << current_dt;
252 pcout << '.' << std::endl;
253 }
254
255 // Advance the time tables in the error estimator
256 if (with_adaptive_timestep)
257 error_estimator->advance(*this);
258
259 rolledback_step = false;
260}

319bool TimeHandler::is_timestep_accepted(
320 LA::ParVectorType &present_solution,
321 const std::vector<LA::ParVectorType> &previous_solutions)
322{
323 using Strategy = Parameters::TimeIntegration::Adaptation::AdaptationStrategy;
324
325 const unsigned int mpi_rank =
326 Utilities::MPI::this_mpi_process(present_solution.get_mpi_communicator());
327
328 // Nonlinear solver failed to find a solution.
329 if (!last_nonlinear_solver_converged)
330 {
331 // First check that it threw correctly from there, if it was needed
332 AssertThrow(!is_steady(), ExcInternalError());
333 AssertThrow(with_adaptive_timestep, ExcInternalError());
334 AssertThrow(!is_starting_step(), ExcInternalError());
335 AssertThrow(!last_step_was_starting_step(), ExcInternalError());
336
337 // Then reject the step
338 if (mpi_rank == 0)
339 std::cout << "Rejecting time step because nonlinear solver failed to "
340 "find a solution"
341 << std::endl;
342 goto reject_step;
343 }
344
345 // Always accept steady-state step or unsteady step without adaptation
346 if (is_steady() || !with_adaptive_timestep)
347 goto accept_step;
348
349 // Also always accept BDF starting steps for now
350 if (is_starting_step() || last_step_was_starting_step())
351 {
352 // Print a warning if the CFL was too high
353 if (time_parameters.adaptation.strategy == Strategy::CFL)
354 {
355 const double cfl_ratio =
356 max_cfl_number / time_parameters.adaptation.target_cfl;
357 if (cfl_ratio > time_parameters.adaptation.reject_cfl_factor)
358 if (mpi_rank == 0)
359 std::cout
360 << "\nWarning : The CFL number at this time steps exceeds the "
361 "ratio for step rejection,\nbut this time step was not rejected "
362 "as it is a starting step for the chosen BDF time\nintegration "
363 "scheme. Consider reducing the initial time step to limit the "
364 "CFL\nnumber during the initial steps.\n"
365 << std::endl;
366 }
367 else if (time_parameters.adaptation.strategy == Strategy::AdaptiveMobility)
368 {
369 const double mobility_ratio =
370 max_adaptive_mobility_number /
371 time_parameters.adaptation.target_adaptive_mobility_number;
372 if (mobility_ratio >
373 time_parameters.adaptation.reject_adaptive_mobility_factor)
374 if (mpi_rank == 0)
375 std::cout
376 << "\nWarning: the adaptive mobility number exceeds the step "
377 "rejection threshold, but this BDF startup step is accepted. "
378 "Consider reducing the initial time step.\n"
379 << std::endl;
380 }
381 goto accept_step;
382 }
383
384 if (time_parameters.adaptation.strategy == Strategy::BDFTruncationError)
385 {
386 // FIXME: Clarify this.
387 // We need bdf_order + 2 solutions to compute the error estimate,
388 // and for methods with starting steps, we also want to start the error
389 // estimation with steps which are not the initial condition.
390 // Thus, start error estimation at step n_starting_steps + bdf_order + 1.
391 const unsigned int n_starting_steps = bdf_order - 1;
392 if (current_time_iteration <= bdf_order + n_starting_steps)
393 return true;
394
395 // Compute the error estimator for this time step
396 error_estimator->compute_error_estimator(*this,
397 present_solution,
398 previous_solutions);
399
400 if (time_parameters.adaptation.reject_timestep_with_large_error)
401 {
402 const auto &max_errors = error_estimator->get_max_errors();
403 const auto &target_errors = time_parameters.adaptation.target_error;
404
405 for (const auto &[variable, error] : max_errors)
406 {
407 Assert(target_errors.count(variable) > 0, ExcInternalError());
408 const double error_ratio = error / target_errors.at(variable);
409
410 if (error_ratio > time_parameters.adaptation.reject_error_factor)
411 {
412 if (mpi_rank == 0)
413 std::cout << "Rejecting step because error ratio for variable \"" +
414 SolverInfo::to_string(variable) + "\" is = "
415 << error_ratio << std::endl;
416 goto reject_step;
417 }
418 }
419 }
420 }
421
422 if (time_parameters.adaptation.strategy == Strategy::CFL)
423 {
424 //
425 if (time_parameters.adaptation.reject_timestep_with_large_cfl)
426 {
427 const double cfl_ratio =
428 max_cfl_number / time_parameters.adaptation.target_cfl;
429
430 if (cfl_ratio > time_parameters.adaptation.reject_cfl_factor)
431 {
432 if (mpi_rank == 0)
433 std::cout << "Rejecting time step because CFL ratio is too large : "
434 << cfl_ratio << " > "
435 << time_parameters.adaptation.target_cfl << std::endl;
436 goto reject_step;
437 }
438 }
439 }
440
441 if (time_parameters.adaptation.strategy == Strategy::AdaptiveMobility)
442 {
443 if (time_parameters.adaptation
444 .reject_timestep_with_large_adaptive_mobility)
445 {
446 const double mobility_ratio =
447 max_adaptive_mobility_number /
448 time_parameters.adaptation.target_adaptive_mobility_number;
449
450 if (mobility_ratio >
451 time_parameters.adaptation.reject_adaptive_mobility_factor)
452 {
453 if (mpi_rank == 0)
454 std::cout << "Rejecting time step because adaptive mobility ratio "
455 "is too large: "
456 << mobility_ratio << " > "
457 << time_parameters.adaptation
458 .reject_adaptive_mobility_factor
459 << std::endl;
460 goto reject_step;
461 }
462 }
463 }
464
465accept_step:
466 // Accepting step : reset counter and update time step
467 n_consecutive_rejected_steps = 0;
468 all_simulation_times.push_back(current_time);
469 increment_inverse_time_step_integral();
470 set_next_timestep(true);
471 return true;
472
473reject_step:
474 rolledback_step = true;
475 n_consecutive_rejected_steps++;
476 n_rejected_steps++;
477
478 // Exit with failure if we already rejected too many steps
479 const unsigned int max_rejections = 5;
480 if (n_consecutive_rejected_steps > max_rejections)
481 {
482 throw std::runtime_error("Could not find an acceptable time step after " +
483 std::to_string(max_rejections) +
484 " step rejections. Aborting simulation.");
485 }
486
487 // Get next time step
488 const double timestep_copy = current_dt;
489 set_next_timestep(false);
490
491 // Rollback the changes from this time step
492 present_solution = previous_solutions[0];
493 current_time -= timestep_copy;
494 current_time_iteration--;
495 current_time_iteration_in_interval--;
496
497 if (mpi_rank == 0)
498 std::cout << "Trying again with time step = " << current_dt << std::endl;
499 return false;
500}

520double clamp_timestep(const double current_timestep,
521 const double next_timestep,
522 const Parameters::TimeIntegration &time_parameters)
523{
524 double clamped_timestep = next_timestep;
525
526 // Bound the increase/decrease ratio
527 double ratio = next_timestep / current_timestep;
528 ratio = std::min(time_parameters.adaptation.max_timestep_increase,
529 std::max(time_parameters.adaptation.max_timestep_reduction,
530 ratio));
531 // Bound by the BDF2 swing factor if needed
532 if (time_parameters.scheme == BDF2)
533 ratio = std::min(1. + sqrt(2.), ratio);
534 clamped_timestep = ratio * current_timestep;
535
536 // Bound the absolute time step
537 clamped_timestep = std::min(time_parameters.adaptation.max_timestep,
538 std::max(time_parameters.adaptation.min_timestep,
539 clamped_timestep));
540
541 return clamped_timestep;
542}

544void TimeHandler::set_next_timestep(const bool step_was_accepted)
545{
546 // BDF starting steps are not modified, even when time step adaptation is
547 // enabled. Set the next time step depending on the current iteration number.
548 if (!with_adaptive_timestep || is_starting_step() ||
549 last_step_was_starting_step())
550 {
551 if (scheme == BDF2 &&
552 time_parameters.bdfstart == Parameters::TimeIntegration::BDFStart::BDF1)
553 {
554 if (is_starting_step())
555 {
556 // Current step was starting step
557 // For next step, apply difference to reach dt
558 current_dt =
559 initial_dt * (1. - time_parameters.bdf_starting_step_ratio);
560 }
561 else if (last_step_was_starting_step())
562 {
563 // Continue simulation with the prescribed constant time step.
564 // If time adaptation is enabled and this step is too large, it may
565 // be rejected starting at the next time step.
566 current_dt = initial_dt;
567 }
568 }
569
570 // Make sure the last time step is the prescribed end time
571 if (current_time_interval == n_time_intervals - 1)
572 current_dt = std::min(current_dt, final_time - current_time);
573
574 return;
575 }
576
577 using Strategy = Parameters::TimeIntegration::Adaptation::AdaptationStrategy;
578
579 double next_timestep = current_dt;
580
581 if (step_was_accepted)
582 {
583 if (time_parameters.adaptation.strategy == Strategy::BDFTruncationError)
584 {
585 const unsigned int n_starting_steps = bdf_order - 1;
586 if (current_time_iteration <= bdf_order + n_starting_steps)
587 return;
588
589 // Error estimator was already computed when checking for whether or not
590 // the last time step should be accepted.
591 // Simply compute the next time step from the stored errors.
592 next_timestep = error_estimator->get_next_timestep(current_dt);
593 }
594 if (time_parameters.adaptation.strategy == Strategy::CFL)
595 {
596 next_timestep =
597 current_dt * time_parameters.adaptation.target_cfl / max_cfl_number;
598 }
599 if (time_parameters.adaptation.strategy == Strategy::AdaptiveMobility)
600 {
601 if (max_adaptive_mobility_number > 0.)
602 next_timestep =
603 current_dt *
604 time_parameters.adaptation.target_adaptive_mobility_number /
605 max_adaptive_mobility_number;
606 else
607 next_timestep = time_parameters.adaptation.max_timestep;
608 }
609
610 // Make sure the last time step is the prescribed end time
611 // FIXME: adjust time step to match the end of the current subinterval
612 if (current_time_interval == n_time_intervals - 1)
613 next_timestep = std::min(next_timestep, final_time - current_time);
614 }
615 else
616 {
617 const double reduction_factor = 0.9;
618
619 if (!last_nonlinear_solver_converged)
620 {
621 // Heuristic for when the nonlinear solver failed
622 // Simply halve the last time step
623 next_timestep = 0.5 * current_dt;
624 }
625 else if (time_parameters.adaptation.strategy ==
626 Strategy::BDFTruncationError)
627 {
628 // Reduce the time step based on the error estimate from this rejected
629 // step, and apply a slight safety factor to avoid multiple rejections
630 // in a row.
631 next_timestep =
632 reduction_factor * error_estimator->get_next_timestep(current_dt);
633 }
634 else if (time_parameters.adaptation.strategy == Strategy::CFL)
635 {
636 // Reduce the time step based on the computed CFL, and apply a slight
637 // safety factor to avoid multiple rejections in a row.
638 next_timestep = reduction_factor * current_dt *
639 time_parameters.adaptation.target_cfl / max_cfl_number;
640 }
641 else if (time_parameters.adaptation.strategy == Strategy::AdaptiveMobility)
642 {
643 // Use the same safety factor and retry policy as CFL adaptation.
644 next_timestep =
645 reduction_factor * current_dt *
646 time_parameters.adaptation.target_adaptive_mobility_number /
647 max_adaptive_mobility_number;
648 }
649 else
650 {
651 DEAL_II_ASSERT_UNREACHABLE();
652 }
653
654 AssertThrow(next_timestep < current_dt, ExcInternalError());
655 }
656
657 // Bound progression ratio and absolute time step
658 next_timestep = clamp_timestep(current_dt, next_timestep, time_parameters);
659
660 // TODO: Do not account for required times right now, as this comes
661 // with a plethora of corner cases to ensure we do not reach tiny time steps
662 // if the next required time is just before/after the next predicted time.
663 // The tiny time steps should be merged, and if not possible, maybe the
664 // accepted time step should be halved, to obtain a time step within bounds
665 // at the next step.
666
667 current_dt = next_timestep;
668}

670double TimeHandler::compute_time_derivative_at_quadrature_node(
671 const unsigned int quadrature_node_index,
672 const double present_solution,
673 const std::vector<std::vector<double>> &previous_solutions) const
674{
675 if (scheme == Parameters::TimeIntegration::Scheme::stationary)
676 return 0.;
677 if (scheme == Parameters::TimeIntegration::Scheme::BDF1 ||
678 scheme == Parameters::TimeIntegration::Scheme::BDF2)
679 {
680 double value_dot = bdf_coefficients[0] * present_solution;
681 for (unsigned int i = 1; i < bdf_coefficients.size(); ++i)
682 value_dot +=
683 bdf_coefficients[i] * previous_solutions[i - 1][quadrature_node_index];
684 return value_dot;
685 }
686 DEAL_II_ASSERT_UNREACHABLE();
687}

689void TimeHandler::write_timestep_history(std::ostream &out) const
690{
691 for (unsigned int i = 0; i < all_simulation_times.size(); ++i)
692 {

713void TimeHandler::update_parameters_after_restart(
714 const Parameters::TimeIntegration &new_parameters)
715{
716 // 'steady_scheme' was loaded from the checkpoint
717 // Save it before overwriting with the new simulation's scheme.
718 const bool is_checkpoint_steady = steady_scheme;
719 steady_scheme = new_parameters.scheme == STAT;
720
721 // Nothing to update if current solver is steady-state
722 if (steady_scheme)
723 return;
724
725 // Detect a restart from a stationary checkpoint into an unsteady simulation.
726 if (is_checkpoint_steady)
727 {
728 // Overwrite the serialized parameters that were loaded from the stationary
729 // checkpoint with their new values.
730 initial_time = new_parameters.t_initial;
731 final_time = new_parameters.t_end;
732 current_time = initial_time;
733 current_time_iteration = 0;
734 current_dt = initial_dt;
735 with_adaptive_timestep = new_parameters.adaptation.enable;
736
737 // Set the initial time step if using BDF1 as BDF2 starting method
738 if (scheme == BDF2 &&
739 new_parameters.bdfstart == Parameters::TimeIntegration::BDFStart::BDF1)
740 {
741 current_dt = initial_dt * new_parameters.bdf_starting_step_ratio;
742 }
743
744 simulation_times.assign(n_previous_solutions + 1, initial_time);
745 time_steps.assign(n_previous_solutions + 1, initial_dt);
746
747 set_bdf_coefficients();
748 return;
749 }
750
751 // For now, both the interrupted and restarted simulation should agree
752 // on whether adaptive time stepping is used.
753 AssertThrow(
754 with_adaptive_timestep == new_parameters.adaptation.enable,
755 ExcMessage(
756 "When restarting a simulation from a checkpoint, both the interrupted "
757 "and the restarted simulations should agree on whether adaptive time "
758 "stepping is used. The parameters used for this restarted simulation do "
759 "not agree with the those from the checkpointed one (one was using "
760 "adaptive time stepping and this one does not, or vice versa)."));
761
762 time_iteration_at_last_restart = current_time_iteration;
763
764 // Update time step and final time, then update BDF coefficients
765 final_time = new_parameters.t_end;
766
767 // If time step adaptation is enabled, keep the predicted time step
768 // stored in the checkpointed data. Otherwise, use the new dt.
769 if (!with_adaptive_timestep)
770 {
771 current_dt = new_parameters.dt;
772 time_steps[0] = new_parameters.dt;
773 }
774
775 set_bdf_coefficients();
776}
