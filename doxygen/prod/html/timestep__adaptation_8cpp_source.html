62void BDFErrorEstimator::compute_error_estimator(
63 const TimeHandler &time_handler,
64 const LA::ParVectorType &present_solution,
65 const std::vector<LA::ParVectorType> &previous_solutions)
66{
67 // Check that attach_data() was called
68 Assert(ordering, ExcDataNotAttached());
69 Assert(locally_relevant_dofs, ExcDataNotAttached());
70 Assert(dofs_to_component, ExcDataNotAttached());
71 AssertDimension(dofs_to_component->size(),
72 locally_relevant_dofs->n_elements());
73
74 Assert(additional_solution.size() > 0, ExcInternalError());
75 Assert(dofs_to_component->size() > 0,
76 ExcMessage("The vector dofs_to_component should be filled before "
77 "entering this function."));
78
79 // If the full vector of error estimates for each dof must be stored and
80 // it has not yet been initialized, do it now
81 if (error_estimator.size() == 0)
82 if (save_full_error_estimator)
83 {
84 locally_owned_elements = present_solution.locally_owned_elements();
85 error_estimator.reinit(locally_owned_elements,
86 *locally_relevant_dofs,
87 present_solution.get_mpi_communicator());
88 fully_distributed_error_estimator.reinit(
89 locally_owned_elements, present_solution.get_mpi_communicator());
90 }
91
92 // Set the handled variables the first time we enter here
93 if (handled_variables.empty())
94 for (const auto var : SolverInfo::variable_types)
95 if (ordering->has_variable(var))
96 {
97 handled_variables.push_back(var);
98 max_error[var] = 0.;
99 }
100
101 Assert(!handled_variables.empty(), ExcInternalError());
102
103 for (const auto var : handled_variables)
104 max_error[var] = 0.;
105
106 const unsigned int n_sol = bdf_order + 2;
107
108 AssertDimension(simulation_times.size(), n_sol);
109 AssertDimension(previous_solutions.size(), bdf_order);
110
111 std::vector<double> times(n_sol), values(n_sol);
112 for (unsigned int i = 0; i < n_sol; ++i)
113 times[i] = simulation_times[i];
114
115 // Compute the coefficient (-1)^{p+1} * sum_i BDF_coeff_i * (t_{n+1} -
116 // t_{n+1-i})^{p+1} multiplying the divided difference
117 const auto &alpha = time_handler.bdf_coefficients;
118 double coeff = 0.;
119 for (unsigned int i = 0; i <= bdf_order; ++i)
120 {
121 const double Hi = simulation_times[0] - simulation_times[i];
122 coeff += alpha[i] * std::pow(Hi, bdf_order + 1);
123 }
124 // FIXME: Probalby useless if taking the abs below
125 coeff *= (bdf_order % 2 == 0) ? -1. : 1.;
126
127 // Current time step
128 const double h = simulation_times[0] - simulation_times[1];
129
130 // Compute trunaction error for each relevant dof
131 for (const auto &dof : *locally_relevant_dofs)
132 {
133 // Solution at times t_{n+1} through t_{n+1-(p+1)}.
134 values[0] = present_solution[dof];
135 for (unsigned int i = 0; i < n_sol - 2; ++i)
136 values[i + 1] = previous_solutions[i][dof];
137 values[n_sol - 1] = additional_solution[dof];
138
139 // Compute the divided difference of order p + 1
140 double dd = 0.;
141 if (time_parameters.scheme == Parameters::TimeIntegration::Scheme::BDF1)
142 dd = divided_difference<2>(times, values);
143 else if (time_parameters.scheme ==
144 Parameters::TimeIntegration::Scheme::BDF2)
145 dd = divided_difference<3>(times, values);
146
147 const double error = std::abs(h * coeff * dd);
148
149 if (save_full_error_estimator)
150 if (locally_owned_elements.is_element(dof))
151 fully_distributed_error_estimator[dof] = error;
152
153 // Get dof component, then variable associated with this component
154 const auto comp =
155 (*dofs_to_component)[locally_relevant_dofs->index_within_set(dof)];
156 const auto var = ordering->component_to_variable_type(comp);
157 max_error.at(var) = std::max(max_error.at(var), error);
158 }
159
160 if (save_full_error_estimator)
161 {
162 fully_distributed_error_estimator.compress(VectorOperation::insert);
163 error_estimator = fully_distributed_error_estimator;
164 }
165
166 // Synchronize the max errors across ranks
167 for (const auto var : handled_variables)
168 {
169 auto *const comm = present_solution.get_mpi_communicator();
170 max_error.at(var) = Utilities::MPI::max(max_error.at(var), comm);
171 if (Utilities::MPI::this_mpi_process(comm) == 0 &&
172 time_parameters.adaptation.verbosity == Parameters::Verbosity::verbose)
173 std::cout << "Max error for variable " + SolverInfo::to_string(var)
174 << " is " << max_error.at(var) << std::endl;
175 }
176}

185double BDFErrorEstimator::get_next_timestep(const double current_timestep) const
186{
187 Assert(!handled_variables.empty(), ExcInternalError());
188 Assert(!max_error.empty(), ExcInternalError());
189
190 std::map<SolverInfo::VariableType, double> target_steps;
191
192 // Compute next time step ratios
193 for (const auto var : handled_variables)
194 {
195 const double eps = time_parameters.adaptation.target_error.at(var);
196 const double ratio =
197 std::pow(eps / std::max(1e-15, max_error.at(var)), 1. / (bdf_order + 1));
198 target_steps[var] = current_timestep * ratio;
199 }
200
201 Assert(!target_steps.empty(), ExcInternalError());
202
203 // Get key and value of minimum timestep
204 const auto it = std::min_element(target_steps.begin(),
205 target_steps.end(),
206 [](const auto &a, const auto &b) {
207 return a.second < b.second;
208 });
209
210 // Return the smallest prescribed timestep
211 return it->second;
212}
