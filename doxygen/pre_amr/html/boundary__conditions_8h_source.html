369 /**
370 * Enforcing zero-mean pressure yields the linear constraint:
371 *
372 * int_\Omega p dx = 0 -> sum_j a_j * p_j = 0,
373 *
374 * where the c_j are obtained from integrating the pressure shape functions
375 * over \Omega. This is satisfied by constraining a single pressure DoF to
376 *
377 * p_0 = - sum_{j != 0} a_j/a_0 * p_j := sum_{j != 0} c_j * p_j.
378 *
379 * This function computes the weights c_j and sets the pressure dof to
380 * constrain "constrained_pressure_dof", which is simply the (globally) first
381 * pressure DoF.
382 *
383 * Important: since this is a global constraint, this pressure dof will be
384 * coupled to *all* other pressure dofs (on MPI processes which have this dof
385 * as owned or ghost). This fills the sparsity pattern and is thus highly
386 * inefficient. Enforcing zero-mean this way is only meant for verification on
387 * corner-case (pun intended, see further) convergence studies, in particular
388 * for 3D tests with some specific non-divergence-free velocity fields, which
389 * show either reduced convergence order for the pressure or even no
390 * convergence at all when setting a corner pressure DoF to zero.
391 *
392 * Also important: Note that this function adds all the non-local pressure
393 * dofs to the vector of locally relevant dofs on ranks for which the
394 * constrained pressure dof is relevant. This should be kept in mind when
395 * performing operations on locally_relevant_dofs. For example, newly added
396 * relevant pressure dofs do not have a matching support point in the map
397 * retrieved from map_dofs_to_support_points.
398 */
399 template <int dim>
400 void create_zero_mean_pressure_constraints_data(
401 const Triangulation<dim> &tria,
402 const DoFHandler<dim> &dof_handler,
403 IndexSet &locally_relevant_dofs,
404 std::vector<unsigned char> &dofs_to_component,
405 const Mapping<dim> &mapping,
406 const Quadrature<dim> &quadrature,
407 const unsigned int p_lower,
408 types::global_dof_index &constrained_pressure_dof,
409 std::vector<std::pair<types::global_dof_index, double>>
410 &constraint_weights);
411
412 /**
413 * Given the weights and dof computed with the function above,
414 * add the single zero-mean constraint on the specified pressure dof to
415 * the passed constraints.
416 *
417 * Important: this yields a very inefficient sparsity pattern and should only
418 * be used for specific verification tests, see above.
419 */
420 void add_zero_mean_pressure_constraints(

573template <typename BCType>
574void BoundaryConditions::declare_boundary_conditions(
575 ParameterHandler &prm,
576 const unsigned int n_bc,
577 const std::string &bc_type_name)
578{
579 // These boundary conditions are used to declare the generic parameters
580 // (problem is that map keys are immutable after they are created, although
581 // we could alternatively move or swap the map entries).
582 std::vector<BCType> tmp_bc(n_bc);
583
584 prm.enter_subsection(bc_type_name + " boundary conditions");
585 {
586 // The number was already parsed in a first dry run
587 prm.declare_entry("number",
588 "0",
589 Patterns::Integer(),
590 "Number of " + bc_type_name + " boundary conditions");
591
592 for (unsigned int i = 0; i < n_bc; ++i)
593 {
594 prm.enter_subsection("boundary " + std::to_string(i));
595 {
596 tmp_bc[i].declare_parameters(prm);
597 }
598 prm.leave_subsection();
599 }
600 }
601 prm.leave_subsection();
602}

604template <typename BCType>
605void BoundaryConditions::read_boundary_conditions(
606 ParameterHandler &prm,
607 const unsigned int n_bc,
608 const std::string &bc_type_name,
609 std::map<types::boundary_id, BCType> &boundary_conditions)
610{
611 /**
612 * FIXME: If the number of bc actually added in the parameter file is
613 * greater than the specified number, deal.ii throws a parse error.
614 * Ideally, we should be able to return an error here if the user specified
615 * too few boundary conditions.
616 */
617 prm.enter_subsection(bc_type_name + " boundary conditions");
618 {
619 for (unsigned int i = 0; i < n_bc; ++i)
620 {
621 prm.enter_subsection("boundary " + std::to_string(i));
622 {
623 unsigned int id = prm.get_integer("id");
624 AssertThrow(
625 id != numbers::invalid_unsigned_int,
626 ExcMessage(
627 bc_type_name + " boundary condition " + std::to_string(i) +
628 " could not be read, possibly because a boundary condition "
629 "sequential number (not the id) is repeated, for instance:\n\n"
630 "subsection boundary 1\n"
631 " set id = 2\n"
632 " set name = boundary_2\n"
633 " set type = type\n"
634 " end\n"
635 "subsection boundary 1 <====== 1 is repeated\n"
636 " set id = 3\n"
637 " set name = boundary_3\n"
638 " set type = type\n"
639 "end"));
640 boundary_conditions[id].read_parameters(prm);
641 }
642 prm.leave_subsection();
643 }
644 }
645 prm.leave_subsection();

648template <int dim, typename VectorType>
649void BoundaryConditions::apply_field_as_solution_on_volume_and_boundaries(
650 const DoFHandler<dim> &dof_handler,
651 const Mapping<dim> &mapping,
652 const Function<dim> &exact_solution,
653 VectorType &present_solution,
654 VectorType &local_present_solution,
655 const IndexSet &locally_relevant_dofs,
656 std::vector<unsigned char> &dofs_to_component,
657 const ComponentMask &component_mask,
658 const bool homogeneous,
659 AffineConstraints<double> &constraints)
660{
661 const auto zero_fun =
662 Functions::ZeroFunction<dim>(exact_solution.n_components);
663 const Function<dim> &f = homogeneous ? zero_fun : exact_solution;
664 VectorTools::interpolate(
665 mapping, dof_handler, f, local_present_solution, component_mask);
666
667 present_solution = local_present_solution;
668
669 if (dofs_to_component.empty())
670 fill_dofs_to_component(dof_handler,
671 locally_relevant_dofs,
672 dofs_to_component);
673
674 for (const auto &dof : locally_relevant_dofs)
675 {
676 const unsigned char comp =
677 dofs_to_component[locally_relevant_dofs.index_within_set(dof)];
678
679 /**
680 * Non-local dofs may have been added to locally_relevant_dofs after its
681 * creation : for these dofs, the component index cannot be determined by
682 * looping over the local cells, thus we cannot decide here if the missing
683 * component index is the field currently being constrained... The map
684 * dofs_to_component should be updated when adding these dofs to
685 * locally_relevant_dofs.
686 *
687 * Currently, this happens when enforcing a zero mean pressure (which adds
688 * all pressure dofs to some ranks), and when adding lambda dofs as relevant
689 * in the FSI solver. The first case is prevented by checking the options in
690 * parameter_reader.cpp, the other should not come up because it does not
691 * make too much sense to set an exact Lagrange multiplier (and there is no
692 * currently no way to do it in the parameter file).
693 */
694 Assert(
695 comp != static_cast<unsigned char>(-1),
696 ExcMessage(
697 "You are trying to apply a prescribed exact field in parallel, but the "
698 "component index for some dofs on this partition could not be "
699 "determined, and "
700 "thus the constraints for this field may be incomplete. This is likely "
701 "because at some point, non-local dofs for this field were added to "
702 "the vector of locally relevant dofs, but their components in the "
703 "dofs_to_component map were not updated accordingly. To solve this, "
704 "the dofs_to_component should be updated whenever ghost dofs are added "
705 "to locally_relevant_dofs."));
706 if (component_mask[comp])
707 if (constraints.can_store_line(dof) && !constraints.is_constrained(dof))
708 constraints.add_constraint(dof, {}, present_solution[dof]);
709 }
710}
