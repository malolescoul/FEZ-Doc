263template <int dim, bool with_moving_mesh>
264void NavierStokesSolver<dim, with_moving_mesh>::run_time_subinterval(
265 const unsigned int interval_index)
266{
267 set_interval_data(interval_index);
268
269 /**
270 * If starting from zero, read the mesh then setup the dof_handler and
271 * vectors. If restarting, setup_dofs() is called in the restart() function,
272 * after loading the mesh.
273 */
274 if (!param.checkpoint_restart.restart)
275 {
276 if (should_create_triangulation())
277 MeshTools::read_mesh(*triangulation, param);
278 setup_dofs();
279 }
280 else
281 {
282 // AMR was not yet tested with restart
283 AssertThrow(!param.with_tree_based_adaptation(),
284 ExcMessage("Simulation restart with adaptive mesh refinement "
285 "(AMR) it currently not implemented."));
286 restart();
287 }
288
289 setup_mappings();
290 initialize_interval(interval_index);
291 create_scratch_data();
292
293 if (param.bc_data.enforce_zero_mean_pressure)
294 create_zero_mean_pressure_constraints_data();
295 create_solver_specific_constraints_data();
296
297 create_zero_constraints();
298 create_nonzero_constraints();
299 create_sparsity_pattern();
300
301 /**
302 * Apply initial refinement.
303 */
304 if (!time_handler.is_steady() && param.with_tree_based_adaptation())
305 {
306 prefix_data.is_prerefinement_step = true;
307 for (unsigned int step = 0;
308 step < param.mesh.adaptation.tree_amr.n_prerefinement_steps;
309 ++step)
310 {
311 update_boundary_conditions();
312 set_initial_conditions(false);
313 adapt_mesh();
314 prefix_data.prerefinement_step = step;
315 output_results();
316 }
317 prefix_data.is_prerefinement_step = false;
318 }
319
320 if (!param.checkpoint_restart.restart)
321 {
322 if (interval_index == 0)
323 set_initial_conditions();
324 else
325 transient_fixed_point_data.transfer_solution_between_intervals(
326 interval_index,
327 *moving_mapping,
328 *exact_solution,
329 time_handler,
330 locally_relevant_dofs,
331 dofs_to_component);
332 }
333
334 // For unsteady simulations, postprocess either the initial condition, or the
335 // initial solution on this time interval. For unsteady simulations with mesh
336 // adaptation with a Riemannian metric, this is needed to obtain an adapted
337 // mesh that includes the initial condition.
338 postprocess_solution();
339
340 while (!time_handler.is_finished())
341 {
342 do
343 {
344 time_handler.advance(pcout);
345 set_time();
346 update_boundary_conditions();
347
348 if (time_handler.is_starting_step() &&
349 param.time_integration.bdfstart ==
350 Parameters::TimeIntegration::BDFStart::initial_condition)
351 {
352 if (param.mms_param.enable || param.debug.apply_exact_solution)
353 // Convergence study: start with exact solution at first time step
354 set_exact_solution();
355 else
356 // Repeat initial condition
357 set_initial_conditions();
358 }
359 else
360 {
361 // Entering the Newton solver with a solution satisfying the nonzero
362 // constraints, which were applied in update_boundary_condition().
363 if (param.nonlinear_solver.compare_jacobian_with_finite_differences)
364 compare_analytical_matrix_with_fd();
365
366 if (param.debug.apply_exact_solution)
367 set_exact_solution();
368 else
369 solve_nonlinear_problem(time_handler);
370 }
371
372 compute_max_cfl();
373 compute_solver_specific_timestep_adaptation_criterion();
374 }
375 while (!time_handler.is_timestep_accepted(*present_solution,
376 *previous_solutions));
377
378 postprocess_solution();
379
380 /**
381 * Adapt the tree-based mesh during an unsteady simulation, if the current
382 * time step iteration matches the prescribed frequency.
383 *
384 * For steady-state simulations, the mesh is adapted after the finalize()
385 * function is called, so that the registered number of mesh elements
386 * and dofs matches the computed error for convergence studies.
387 */
388 if (should_adapt_tree_based_mesh(time_handler))
389 adapt_mesh();
390
391 time_handler.rotate_solutions(*present_solution, *previous_solutions);
392
393 if (param.checkpoint_restart.enable_checkpoint &&
394 (time_handler.current_time_iteration %
395 param.checkpoint_restart.checkpoint_frequency ==
396 0))
397 {
398 /**
399 * Write checkpoint.
400 * Checkpoint is written *after* the solutions were rotated, which amounts
401 * to writing the current solution twice... For some applications or
402 * postprocessing, maybe it would be useful to checkpoint before rotating,
403 * to actually save the last N solutions. In that case, the solutions
404 * would need to be rotated right after restart().
405 */
406 checkpoint();
407 }
408 }
409
410 finalize_interval(interval_index);
411}

413template <int dim, bool with_moving_mesh>
414void NavierStokesSolver<dim, with_moving_mesh>::run()
415{
416 reset();
417 initialize();
418
419 for (unsigned int i = 0; i < param.time_integration.n_time_intervals; ++i)
420 run_time_subinterval(i);
421
422 finalize();
423
424 /**
425 * If using a riemannian metric to adapt the mesh(es), perform all the
426 * adaptations at the end of all time intervals (as it requires a global
427 * scaling factor).
428 *
429 * If using tree-based adaptation with a steady-state convergence study,
430 * adapt the mesh here.
431 */
432 if (should_scale_and_grade_riemannian_metric(param, time_handler))
433 {
434 transient_fixed_point_data.scale_metrics(
435 param.metrics.metric_for_adaptation, time_handler);
436 transient_fixed_point_data.apply_gradation_to_metrics();
437 }
438 if (should_adapt_mesh_at_end_of_intervals(time_handler))
439 adapt_mesh();
440}

483template <int dim, bool with_moving_mesh>
484void NavierStokesSolver<dim, with_moving_mesh>::setup_mappings()
485{
486 TimerOutput::Scope t(computing_timer, "Setup mappings");
487
488 if constexpr (with_moving_mesh)
489 {
490 // Initialize mesh position directly from the triangulation.
491 // The parallel vector storing the mesh position is local_evaluation_point,
492 // because this is the one to modify when computing finite differences.
493 // If the solution was restarted, the evaluation point already stores
494 // the current solution, so don't overwrite it.
495 if (!param.checkpoint_restart.restart)
496 {
497 VectorTools::get_position_vector(*fixed_mapping,
498 *dof_handler,
499 local_evaluation_point,
500 position_mask);
501 local_evaluation_point.compress(VectorOperation::insert);
502 evaluation_point = local_evaluation_point;
503 }
504
505 // Also store them in initial_positions, for postprocessing:
506 initial_positions = DoFTools::map_dofs_to_support_points(*fixed_mapping,
507 *dof_handler,
508 position_mask);
509
510 // Create the solution-dependent mapping
511 moving_mapping =
512 std::make_unique<MappingFEField<dim, dim, LA::ParVectorType>>(
513 *dof_handler, evaluation_point, position_mask);
514 }
515 else
516 {
517 // Moving_mapping and fixed_mapping are identical
518 if (param.finite_elements.use_quads)
519 moving_mapping =
520 std::make_unique<MappingQ<dim>>(param.finite_elements.mapping_degree);
521 else
522 moving_mapping = std::make_unique<MappingFE<dim>>(
523 FE_SimplexP<dim>(param.finite_elements.mapping_degree));
524 }

597template <int dim, bool with_moving_mesh>
598void NavierStokesSolver<dim, with_moving_mesh>::create_base_constraints(
599 const bool homogeneous,
600 AffineConstraints<double> &constraints)
601{
602 constraints.clear();
603 constraints.reinit(locally_owned_dofs, locally_relevant_dofs);
604
605 if (param.with_tree_based_adaptation())
606 DoFTools::make_hanging_node_constraints(*dof_handler, constraints);
607
608 /**
609 * Set whole field from exact solution if required, and add the associated
610 * constraints for the volume and boundary dofs.
611 *
612 * Do this before applying other constraints (e.g., fluxes).
613 */
614 for (const auto &[field_name, mask] : field_names_and_masks)
615 {
616 if (param.mms.set_field_as_solution.at(field_name))
617 {
618 /**
619 * Setting mesh position first is already accounted for in
620 * update_boundary_conditions(). During the second run, the moving mapping
621 * has been updated with the exact position at the current time, and the
622 * other fields can be constrained based on the exact solution evaluated
623 * on the moving mapping.
624 */
625 BoundaryConditions::apply_field_as_solution_on_volume_and_boundaries(
626 *dof_handler,
627 field_name == "mesh position" ? *fixed_mapping : *moving_mapping,
628 *exact_solution,
629 evaluation_point,
630 local_evaluation_point,
631 locally_relevant_dofs,
632 dofs_to_component,
633 mask,
634 homogeneous,
635 constraints);
636 }
637 }
638
639 /**
640 * If relevant, apply mesh boundary conditions first, as they affect
641 * the evaluation of the fields on the moving mesh.
642 */
643 if constexpr (with_moving_mesh)
644 {
645 BoundaryConditions::apply_mesh_position_boundary_conditions(
646 homogeneous,
647 ordering->x_lower,
648 ordering->n_components,
649 *dof_handler,
650 *fixed_mapping,
651 param.pseudosolid_bc,
652 *exact_solution,
653 *param.mms.exact_mesh_position,
654 constraints);
655 }
656
657 BoundaryConditions::apply_velocity_boundary_conditions(
658 homogeneous,
659 ordering->u_lower,
660 ordering->n_components,
661 *dof_handler,
662 *moving_mapping,
663 param.fluid_bc,
664 *exact_solution,
665 *param.mms.exact_velocity,
666 constraints);
667
668 if (param.bc_data.fix_pressure_constant)
669 {
670 // The pressure DOF is set to 0 by default for the nonzero constraints too,
671 // unless there is a prescribed manufactured solution, in which case it is
672 // prescribed to p_mms.
673 bool set_to_zero = true;
674 if (!homogeneous && param.mms_param.enable)
675 set_to_zero = false;
676
677 if constexpr (with_moving_mesh)
678 {
679 // Update the location of the support point of constrained pressure dof
680 // FIXME: this calls map_dofs_to_support_points, but is only done for MMS
681 if (!set_to_zero &&
682 constrained_pressure_dof != numbers::invalid_dof_index &&
683 locally_relevant_dofs.is_element(constrained_pressure_dof))
684 {
685 const auto support_points =
686 DoFTools::map_dofs_to_support_points(*moving_mapping, *dof_handler);
687 constrained_pressure_support_point =
688 support_points.at(constrained_pressure_dof);
689 }
690 }
691
692 BoundaryConditions::constrain_pressure_point(
693 *dof_handler,
694 locally_relevant_dofs,
695 *moving_mapping,
696 *exact_solution,
697 ordering->p_lower,
698 set_to_zero,
699 constraints,
700 constrained_pressure_dof,
701 constrained_pressure_support_point);
702 }
703
704 if (param.bc_data.enforce_zero_mean_pressure)
705 BoundaryConditions::add_zero_mean_pressure_constraints(
706 constraints,
707 locally_relevant_dofs,
708 constrained_pressure_dof,
709 zero_mean_pressure_weights);
710
711 // FIXME: group all pressure conditions in this function
712 BoundaryConditions::apply_pressure_boundary_conditions(homogeneous,
713 ordering->p_lower,
714 ordering->n_components,
715 *dof_handler,
716 *moving_mapping,
717 param.fluid_bc,
718 *exact_solution,
719 constraints);
720 /**
721 * Do not close the constraints here, as derived solvers may need
722 * to add boundary conditions on their own fields (e.g., Cahn Hilliard)

743template <int dim, bool with_moving_mesh>
744void NavierStokesSolver<dim, with_moving_mesh>::set_initial_conditions(
745 const bool rotate_solutions)
746{
747 /**
748 * Mesh position should be evaluated and updated *BEFORE* evaluating fields on
749 * moving mapping. This matters in the rare cases when the initial mesh
750 * position is *not* the fixed_mapping.
751 */
752
753 const Function<dim> *velocity_fun =
754 param.initial_conditions.set_to_mms ?
755 exact_solution.get() :
756 param.initial_conditions.initial_velocity.get();
757
758 if constexpr (with_moving_mesh)
759 {
760 FixedMeshPosition<dim> fixed_mesh(ordering->x_lower,
761 ordering->n_components);
762
763 const Function<dim> *mesh_fun =
764 param.initial_conditions.set_to_mms ? exact_solution.get() : &fixed_mesh;
765
766 // Set mesh position with fixed mapping
767 VectorTools::interpolate(
768 *fixed_mapping, *dof_handler, *mesh_fun, newton_update, position_mask);
769
770 // Update MappingFEField *BEFORE* interpolating velocity
771 evaluation_point = newton_update;
772
773 // Inject the presolved mesh position (if a presolver is attached) before
774 // interpolating the other fields, so that they are evaluated on the
775 // presolved moving mesh.
776 if (presolver != nullptr)
777 {
778 *present_solution = newton_update;
779 overwrite_position_from_presolver(*presolver);
780 newton_update = *present_solution;
781 evaluation_point = newton_update;
782 }
783 }
784
785 // Set velocity with moving mapping
786 VectorTools::interpolate(
787 *moving_mapping, *dof_handler, *velocity_fun, newton_update, velocity_mask);
788
789 // Set other solver-specific fields on moving mesh (e.g., CHNS tracer)
790 set_solver_specific_initial_conditions();
791
792 // Constraints created during setup were evaluated on the undeformed mesh.
793 // Recompute their values now that the initial ALE mapping (including the
794 // presolved mesh position) has been initialized.
795 if constexpr (with_moving_mesh)
796 create_nonzero_constraints();
797
798 // Apply non-homogeneous Dirichlet BC and set as current solution
799 nonzero_constraints.distribute(newton_update);
800 *present_solution = newton_update;
801 evaluation_point = newton_update;
802
803 if (rotate_solutions)
804 // FIXME: WHAT ABOUT THIS ROTATION?????????
805 time_handler.rotate_solutions(*present_solution, *previous_solutions);
806}

808template <int dim, bool with_moving_mesh>
809void NavierStokesSolver<dim, with_moving_mesh>::overwrite_position_from_presolver(
810 ElasticitySolver<dim> &presolver)
811{
812 local_evaluation_point = *present_solution;
813
814 // The presolver solves the elasticity (mesh-position) system alone, whose
815 // components 0..dim-1 map to this solver's mesh-position components.
816 std::map<unsigned int, unsigned int> component_map;
817 for (unsigned int d = 0; d < dim; ++d)
818 component_map[d] = ordering->x_lower + d;
819
820 // When the enlarged marker psi is present in both the presolver and this
821 // solver, also inject the presolver's equilibrium psi as the CHNS initial
822 // guess. The presolved (compressed) mesh is in equilibrium with the forcing
823 // built from that psi; starting the CHNS with psi = 0 would leave the
824 // pseudosolid residual = elastic stress of the compressed mesh unbalanced,
825 // blowing up the very first residual at the handoff. The presolver's psi is
826 // its component dim (ComponentOrderingElasticity appends psi after position).
827 const bool presolver_has_psi =
828 presolver.get_dof_handler().get_fe().n_components() > dim;
829 const bool this_has_psi =
830 ordering->has_variable(SolverInfo::VariableType::phase_enlarged);
831 if (presolver_has_psi && this_has_psi)
832 component_map[dim] = ordering->psi_lower;
833
834 extract_subsolution<dim>(presolver.get_dof_handler(),
835 *dof_handler,
836 presolver.get_present_solution(),
837 local_evaluation_point,
838 component_map);
839 local_evaluation_point.compress(VectorOperation::insert);
840
841 *present_solution = local_evaluation_point;
842 evaluation_point = *present_solution;
843}

999template <int dim, bool with_moving_mesh>
1000void NavierStokesSolver<dim, with_moving_mesh>::compute_errors()
1001{
1002 TimerOutput::Scope t(this->computing_timer, "Compute errors");
1003
1004 const unsigned int n_components = ordering->n_components;
1005 const unsigned int u_lower = ordering->u_lower;
1006 const unsigned int p_lower = ordering->p_lower;
1007
1008 const unsigned int n_active_cells = triangulation->n_active_cells();
1009 Vector<double> cellwise_errors(n_active_cells);
1010
1011 /**
1012 * Set the function pointer to use, depending on whether the mean pressure
1013 * should be subtracted or not.
1014 */
1015 std::shared_ptr<Function<dim>> used_exact_solution = exact_solution;
1016
1017 if (param.bc_data.enforce_zero_mean_pressure)
1018 {
1019 // Mean pressure value
1020 const double p_mean = VectorTools::compute_mean_value(
1021 *moving_mapping, *dof_handler, *quadrature, *present_solution, p_lower);
1022
1023 AssertThrow(std::abs(p_mean) < 1e-10,
1024 ExcMessage(
1025 "Mean pressure should be zero, but it's not : p_mean = " +
1026 std::to_string(p_mean)));
1027
1028 if (param.mms_param.enable)
1029 {
1030 // Use a function wrapper where the pressure mean is subtracted
1031 const double p_mms_mean = compute_global_mean_value(*exact_solution,
1032 p_lower,
1033 *dof_handler,
1034 *moving_mapping);
1035
1036 if (param.mms_param.subtract_mean_pressure)
1037 used_exact_solution =
1038 std::make_shared<PressureMeanSubtractedFunction<dim>>(*exact_solution,
1039 p_mms_mean,
1040 p_lower);
1041 else
1042 // Use the manufactured pressure which is then assumed to have zero
1043 // mean. Throw an error if it's not the case.
1044 AssertThrow(
1045 std::abs(p_mms_mean) < 1e-6,
1046 ExcMessage(
1047 "You are comparing a discrete zero-mean pressure with a "
1048 "manufactured "
1049 "pressure which is not zero-mean. The mean exact pressure is " +
1050 std::to_string(p_mms_mean)));
1051 }
1052 }
1053
1054 /**
1055 * Compute errors on velocity, pressure and position if applicable
1056 */
1057 const ComponentSelectFunction<dim> velocity_comp_select(
1058 std::make_pair(u_lower, u_lower + dim), n_components);
1059 const ComponentSelectFunction<dim> pressure_comp_select(p_lower,
1060 n_components);
1061
1062 compute_and_add_errors(*moving_mapping,
1063 *used_exact_solution,
1064 cellwise_errors,
1065 velocity_comp_select,
1066 "u");
1067 compute_and_add_errors(*moving_mapping,
1068 *used_exact_solution,
1069 cellwise_errors,
1070 pressure_comp_select,
1071 "p");
1072 if constexpr (with_moving_mesh)
1073 {
1074 // Error on mesh position
1075 const unsigned int x_lower = ordering->x_lower;
1076 const ComponentSelectFunction<dim> position_comp_select(
1077 std::make_pair(x_lower, x_lower + dim), n_components);
1078 compute_and_add_errors(*fixed_mapping,
1079 *exact_solution,
1080 cellwise_errors,
1081 position_comp_select,
1082 "x");
1083 }
1084
1085 compute_solver_specific_errors();
1086}

1385template <int dim, bool with_moving_mesh>
1386template <class Archive>
1387void NavierStokesSolver<dim, with_moving_mesh>::load(
1388 Archive &ar,
1389 const unsigned int /* version */)
1390{
1391 ar &(*present_solution);
1392 present_solution->update_ghost_values();
1393
1394 unsigned int n_previous_solutions;
1395 ar &n_previous_solutions;
1396
1397 // Allow restarting an unsteady simulation from a stationary checkpoint: the
1398 // stationary checkpoint contains no previous solutions, so the present
1399 // (steady) solution is duplicated into all slots of the unsteady previous
1400 // solutions array to serve as the initial condition for the unsteady run.
1401 if (n_previous_solutions == 0 && !time_handler.is_steady())
1402 {
