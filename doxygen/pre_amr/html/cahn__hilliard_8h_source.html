10 // Internal, dimensionless regularization constants. The activation scales
11 // are derived from the last tanh tail level that the correction must retain;
12 // they are deliberately not user parameters.
13 constexpr double profile_correction_tail_level = 0.999;
14 constexpr double profile_correction_beta = 0.05;

16 inline double quintic_transition(const double value,
17 const double lower,
18 const double upper)
19 {
20 if (value <= lower)
21 return 0.;
22 if (value >= upper)
23 return 1.;
24
25 const double x = (value - lower) / (upper - lower);
26 return x * x * x * (10. + x * (-15. + 6. * x));
27 }
28
29 inline double quintic_transition_derivative(const double value,
30 const double lower,
31 const double upper)
32 {
33 if (value <= lower || value >= upper)
34 return 0.;
35
36 const double x = (value - lower) / (upper - lower);
37 return 30. * x * x * (x - 1.) * (x - 1.) / (upper - lower);
38 }

46 /** Activation declaring that phi still belongs to the retained tanh tail.
47 * It is C2, zero in pure phases and for every overshoot |phi| > 1. */
48 inline double profile_correction_phase_activation(const double tracer)
49 {
50 const double s = 1. - tracer * tracer;
51 const double s_tail = profile_correction_tail_scale();
52 return quintic_transition(s, 0.25 * s_tail, 0.5 * s_tail);
53 }

55 inline double
56 profile_correction_phase_activation_derivative(const double tracer)
57 {
58 const double s = 1. - tracer * tracer;
59 const double s_tail = profile_correction_tail_scale();
60 return -2. * tracer *
61 quintic_transition_derivative(s, 0.25 * s_tail, 0.5 * s_tail);
62 }

88 /** Li profile driver with a q-relative regularization. beta_PC is purely a
89 * Newton regularization and is not part of the physical Li formulation. */
90 template <int dim>
91 inline dealii::Tensor<1, dim>
92 profile_correction_flux_driver(
93 const double tracer,
94 const dealii::Tensor<1, dim> &tracer_gradient,
95 const double epsilon)
96 {
97 const double phase_activation =
98 profile_correction_phase_activation(tracer);
99 if (phase_activation == 0.)
100 return {};
101
102 const double q =
103 (1. - tracer * tracer) / (std::sqrt(2.) * epsilon);
104 const double R = std::sqrt(
105 tracer_gradient.norm_square() +
106 profile_correction_beta * profile_correction_beta * q * q);
107 const double Q =
108 std::sqrt(1. + profile_correction_beta * profile_correction_beta) * q;
109 return phase_activation * (1. - Q / R) * tracer_gradient;
110 }
111
112 template <int dim>
113 inline dealii::Tensor<1, dim>
114 profile_correction_flux_driver_variation(
115 const double tracer,
116 const double tracer_variation,
117 const dealii::Tensor<1, dim> &tracer_gradient,
118 const dealii::Tensor<1, dim> &tracer_gradient_variation,
119 const double epsilon)
120 {
121 const double phase_activation =
122 profile_correction_phase_activation(tracer);
123 if (phase_activation == 0.)
124 return {};
125
126 const double phase_activation_variation =
127 profile_correction_phase_activation_derivative(tracer) *
128 tracer_variation;
129 const double q =
130 (1. - tracer * tracer) / (std::sqrt(2.) * epsilon);
131 const double q_variation =
132 -std::sqrt(2.) * tracer * tracer_variation / epsilon;
133 const double beta_squared =
134 profile_correction_beta * profile_correction_beta;
135 const double R = std::sqrt(tracer_gradient.norm_square() +
136 beta_squared * q * q);
137 const double R_variation =
138 (tracer_gradient * tracer_gradient_variation +
139 beta_squared * q * q_variation) /
140 R;
141 const double sqrt_one_plus_beta_squared =
142 std::sqrt(1. + beta_squared);
143 const double Q = sqrt_one_plus_beta_squared * q;
144 const double Q_variation = sqrt_one_plus_beta_squared * q_variation;
145 const double factor = 1. - Q / R;
146 const double factor_variation =
147 -Q_variation / R + Q * R_variation / (R * R);
148
149 return (phase_activation_variation * factor +
150 phase_activation * factor_variation) *
151 tracer_gradient +
152 phase_activation * factor * tracer_gradient_variation;
153 }

155 /** Weighted true unit normal used only by the Soligo flux correction. */
156 template <int dim>
157 inline dealii::Tensor<1, dim> flux_correction_normal(
158 const Parameters::CahnHilliard<dim> &param,
159 const double tracer,
160 const dealii::Tensor<1, dim> &tracer_gradient)
161 {
162 const double phase_activation =
163 profile_correction_phase_activation(tracer);
164 if (phase_activation == 0.)
165 return {};
166
167 const double gradient_norm = tracer_gradient.norm();
168 const double gradient_activation =
169 flux_correction_gradient_activation(param, gradient_norm);
170 if (gradient_activation == 0.)
171 return {};
172
173 return phase_activation * gradient_activation * tracer_gradient /

186 // --- CHNS model selection -------------------------------------------------
187 // Two diffuse-interface models share the same unknowns (u, p, phi, mu) but
188 // differ in the potential scaling, the capillary momentum force and the
189 // presence of diffusive inertia:
190 // * abels : mu = sigma_tilde/eps phi(phi^2-1) - sigma_tilde eps
191 // lap(phi); capillary force phi*grad(mu); diffusive
192 // inertia present.
193 // * ding_horriche : mu = phi(phi^2-1) - eps^2 lap(phi) (unscaled
194 // potential); capillary force gamma*mu*grad(phi); no
195 // diffusive inertia.

255 inline dealii::Tensor<1, dim> phase_diffusion_flux_driver(

276 /** Complete directional derivative of K_phi=-J_phi. The weighted FC normal
277 * and |grad(phi)| are supplied by the quadrature-point assembly so they are
278 * computed only once and reused for every trial function. */
279 template <int dim>
280 inline dealii::Tensor<1, dim> phase_diffusion_flux_driver_variation(
281 const Parameters::CahnHilliard<dim> &param,
282 const double tracer,
283 const double tracer_variation,
284 const dealii::Tensor<1, dim> &tracer_gradient,
285 const dealii::Tensor<1, dim> &tracer_gradient_variation,
286 const dealii::Tensor<1, dim> &potential_gradient,
287 const dealii::Tensor<1, dim> &potential_gradient_variation,
288 const double mobility,
289 const double mobility_variation,
290 const dealii::Tensor<1, dim> &normal,
291 const double gradient_norm)
292 {
293 if (!has_interface_profile_correction(param))
294 return mobility_variation * potential_gradient +
295 mobility * potential_gradient_variation;
296
297 auto chemical_gradient = potential_gradient;
298 auto chemical_gradient_variation = potential_gradient_variation;
299 if (has_interface_flux_correction(param))
300 {
301 dealii::Tensor<1, dim> normal_variation;
302 const double phase_activation =
303 profile_correction_phase_activation(tracer);
304 const double gradient_activation =
305 flux_correction_gradient_activation(param, gradient_norm);
306 if (phase_activation > 0. && gradient_activation > 0.)
307 {
308 const auto unit_normal = tracer_gradient / gradient_norm;
309 const double norm_variation =
310 unit_normal * tracer_gradient_variation;
311 const auto unit_normal_variation =
312 (tracer_gradient_variation -
313 unit_normal * (unit_normal * tracer_gradient_variation)) /
314 gradient_norm;
315 const double phase_activation_variation =
316 profile_correction_phase_activation_derivative(tracer) *
317 tracer_variation;
318 const double gradient_activation_variation =
319 flux_correction_gradient_activation_derivative(
320 param, gradient_norm) *
321 norm_variation;
322 const double weight = phase_activation * gradient_activation;
323 const double weight_variation =
324 gradient_activation * phase_activation_variation +
325 phase_activation * gradient_activation_variation;
326 normal_variation = weight_variation * unit_normal +
327 weight * unit_normal_variation;
328 }
329
330 chemical_gradient =
331 project_chemical_potential_gradient<dim>(potential_gradient, normal);
332 chemical_gradient_variation =
333 project_chemical_potential_gradient<dim>(
334 potential_gradient_variation, normal) -
335 normal_variation * (normal * potential_gradient) -
336 normal * (normal_variation * potential_gradient);
337 }
338
339 const auto profile_driver = profile_correction_flux_driver<dim>(
340 tracer, tracer_gradient, param.epsilon_interface);
341 const auto profile_driver_variation =
342 profile_correction_flux_driver_variation<dim>(
343 tracer,
344 tracer_variation,
345 tracer_gradient,
346 tracer_gradient_variation,
347 param.epsilon_interface);
348 const double kappa = profile_correction_coefficient(param, mobility);
349 const double kappa_variation =
350 profile_correction_coefficient(param, mobility_variation);
351
352 return mobility_variation * chemical_gradient +
353 mobility * chemical_gradient_variation +
354 kappa_variation * profile_driver +
355 kappa * profile_driver_variation;
356 }

517 * Static wetting condition for a tanh diffuse interface:
518 *
519 * n . grad(phi) = -cos(theta) (1 - phi^2) / (sqrt(2) eps),

659 template <int dim>
660 inline MobilityEvaluation<dim>
661 evaluate_adaptative_mobility(const Parameters::CahnHilliard<dim> &param,

689 /** Restrict the adaptive-mobility sensor to the interface core.
690 *
691 * The weight is one in the core, decreases with a complementary quintic
692 * smoothstep between |phi|=0.5 and |phi|=0.9, and vanishes in the outer
693 * tails. Using phi^2 makes the weight even; the smoothstep makes both joins
694 * C2-compatible.
695 */
696 inline AdaptiveMobilityRestrictionWeight
697 evaluate_adaptive_mobility_2_restriction_weight(const double phi)
698 {
699 constexpr double phi_core = 0.5;
700 constexpr double phi_cutoff = 0.9;
701 constexpr double r_core = phi_core * phi_core;
702 constexpr double r_cutoff = phi_cutoff * phi_cutoff;
703
704 const double r = phi * phi;
705 if (r <= r_core)
706 return {1., 0., 0.};
707 if (r >= r_cutoff)
708 return {0., 0., 0.};
709
710 const double width = r_cutoff - r_core;
711 const double t = (r - r_core) / width;
712 const double t2 = t * t;
713 const double t3 = t2 * t;
714 const double t4 = t3 * t;
715 const double t5 = t4 * t;
716 const double smoothstep = 6. * t5 - 15. * t4 + 10. * t3;
717 const double smoothstep_d_t = 30. * t2 * (t - 1.) * (t - 1.);
718 const double smoothstep_dd_t2 =
719 60. * t * (2. * t2 - 3. * t + 1.);
720 const double weight = 1. - smoothstep;
721 const double weight_d_r = -smoothstep_d_t / width;
722 const double weight_dd_r2 = -smoothstep_dd_t2 / (width * width);
723
724 return {weight,
725 2. * phi * weight_d_r,
726 2. * weight_d_r + 4. * r * weight_dd_r2};
727 }

729 template <int dim>
730 inline MobilityEvaluation<dim>
731 evaluate_adaptative_mobility_2(const Parameters::CahnHilliard<dim> &,
732 const double phi,
733 const double phi_d,
734 const double phi_dd,
735 const dealii::Tensor<1, dim> &velocity,
736 const dealii::Tensor<1, dim> &tracer_gradient,
737 const double adaptive_coefficient,
738 const double delta)
739 {
740 const auto weight =
741 evaluate_adaptive_mobility_2_restriction_weight(phi);
742 const double weight_d = weight.first_derivative * phi_d;
743 const double weight_dd = weight.second_derivative * phi_d * phi_d +
744 weight.first_derivative * phi_dd;
745 const double velocity_dot_gradient = velocity * tracer_gradient;
746 const double raw = adaptive_coefficient * weight.value *
747 velocity_dot_gradient;
748 const double raw_d = adaptive_coefficient * weight_d *
749 velocity_dot_gradient;
750 const double raw_dd = adaptive_coefficient * weight_dd *
751 velocity_dot_gradient;
752 const double value = std::sqrt(raw * raw + delta * delta);
753 const double derivative = raw * raw_d / value;
754 const double second_derivative =
755 (raw_d * raw_d + raw * raw_dd) / value -
756 (raw * raw_d) * (raw * raw_d) / (value * value * value);
757 const double sensitivity =
758 raw / value * adaptive_coefficient * weight.value;
759 return {value, derivative, second_derivative, sensitivity};
760 }

874 // --- Material phase marker m(phi) -----------------------------------------
875 // The material properties (density, viscosity) and the transported/conserved
876 // variable are affine in a material marker m(phi):
877 // * abels / ding_horriche : m = phi (m' = 1, m'' = 0)
878 // * abels_nlm : m = q = tanh(k phi)/tanh(k)
879 // For abels_nlm this sharpens the material transition (large k) while phi
880 // keeps the capillary energy. The solved unknowns stay (u, p, phi, mu):
881 // - phi is the DOF (NOT q): q = s_k^{-1} inverse is atanh, undefined at
882 // q = +/-1, so phi would blow up in the bulk on any over/undershoot.
883 // - mu is the potential conjugate to q (mu_q), NOT mu_phi: the potential
884 // equation then reads s'_k(phi) mu = mu_phi(phi), i.e. s'_k always
885 // MULTIPLIES; taking mu_phi as the DOF would need mu_q = mu_phi / s'_k in
886 // the transport flux, i.e. a DIVISION by s'_k (which vanishes far from
887 // the interface) -> blow-up. See material_phase helpers below.
888
889 // tanh mixing from val_a (marker = 1) to val_b (marker = -1); with
890 // (val_a, val_b) = (1, -1) this returns q = tanh(k phi)/tanh(k).
891 inline double tanh_mixing(const double phase_marker,
892 const double val_a,
893 const double val_b,
894 const double k)
895 {
896 const double tanh_k = std::tanh(k);
897 const double tanh_phi = std::tanh(k * phase_marker);
898 return ((tanh_k + tanh_phi) * val_a + (tanh_k - tanh_phi) * val_b) /
899 (2. * tanh_k);
900 }
901
902 // Derivative w.r.t. the tracer of tanh_mixing.
903 inline double tanh_mixing_derivative(const double phase_marker,
904 const double val_a,
905 const double val_b,
906 const double k)
907 {
908 const double tanh_k = std::tanh(k);
909 const double tanh_phi = std::tanh(k * phase_marker);
910 const double sech2_phi = 1. - tanh_phi * tanh_phi;
911 return 0.5 * (val_a - val_b) * k / tanh_k * sech2_phi;
912 }
913
914 // Second derivative w.r.t. the tracer of tanh_mixing.
915 inline double tanh_mixing_second_derivative(const double phase_marker,
916 const double val_a,
917 const double val_b,
918 const double k)
919 {
920 const double tanh_k = std::tanh(k);
921 const double tanh_phi = std::tanh(k * phase_marker);
922 const double sech2_phi = 1. - tanh_phi * tanh_phi;
923 return -(val_a - val_b) * k * k / tanh_k * sech2_phi * tanh_phi;
924 }
