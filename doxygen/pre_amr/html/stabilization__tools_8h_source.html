18 /**
19 * Compute the *square* of the cell length used to define the stabilization
20 * parameter tau_SUPG, since this length only intervenes through its square in
21 * the definition of tau_SUPG.
22 *
23 * In two dimensions, we use Tezduyar's definition which estimates the length
24 * of the cell
25 * in the direction of the @p convective velocity u:
26 *
27 * h := 2 * ||u|| / (sum_i |u \cdot grad phi_i|),
28 *
29 * where the phi_i are the isoparametric, scalar-valued shape functions of the
30 * field to stabilize.
31 *
32 * FIXME/ONGOING: This definition seems not well suited for 3D anisotropic
33 * elements. For instance, consider a velocity field u = (1, 0, 1) and the
34 * tetrahedra parameterized by the constant "a" and described by:
35 *
36 * t1 = (0 0 0), (1 0 0), (1 a 0), (1 a 1)
37 * t2 = (0 0 0), (1 0 0), (1 a 1), (1 0 1).
38 *
39 * This definition yields h_1 = 1, but h_2 = sqrt(2), and these values are
40 * unchanged even though both triangles become the same as a goes to zero.
41 *
42 * In three dimensions, the cell length is defined as its diameter, and is
43 * computed using deal.II's diameter() function (longest diagonal for
44 * quads/hexes and longest side for simplices).
45 *
46 * FIXME: This does not handle anisotropic elements well, as it simply returns
47 * the longest side.
48 */
49 template <int dim>
50 inline double
51 compute_h_tau_squared(const unsigned int n_dofs_per_cell,
52 const double cell_diameter,
53 const Tensor<1, dim> &convective_velocity,
54 const double velocity_norm_squared,
55 const std::vector<Tensor<1, dim>> &grad_scalar_shape);

57 /**
58 * Compute the *square* of the cell length used to define the stabilization
59 * parameter according to Tezduyar:
60 *
61 * h_squared := (2 * ||u|| / (sum_i |u \cdot grad phi_i|))^2.
62 */
63 template <int dim>
64 double compute_squared_cell_length_along_velocity_field(
65 const unsigned int n_dofs_per_cell,
66 const double cell_diameter,
67 const Tensor<1, dim> &convective_velocity,
68 const double velocity_norm_squared,
69 const std::vector<Tensor<1, dim>> &grad_scalar_shape);

71 /**
72 * Compute the stabilization parameter tau_SUPG according to Tezduyar's
73 * definition, including the modification for higher-order elements from
74 * Saavedra, L. P., Munch, P., & Blais, B. (2025). A matrix-free stabilized
75 * solver for the incompressible Navier-Stokes equations. This writes:
76 *
77 * / 2 2 2 \ -1/2
78 * | / 1 \ / 2 ||u|| p \ / 4 diff p^2 \ |
79 * tau_SUPG = | |----| + | ----------- | + 9 | ----------- | | ,
80 * | \ dt / \ h_conv / \ h_diff^2 / |
81 * \ /
82 *
83 * where dt is the current time step, u is the @p convective_velocity, p is
84 * the polynomial @p degree of the field to stabilize, h_conv = h_diff are
85 * taken to be identical and given by the result of compute_h_tau_squared(),
86 * and diff is the characteristic @p diffusivity coefficient of the convection-
87 * diffusion problem of interest.
88 *
89 * For steady-state computations, the first term is ignored.
90 *
91 * This definition differs from Garon & Fortin's definition in the coefficient
92 * in the unsteady term.
93 *
94 */
95 template <int dim>
96 inline double
97 compute_tau_supg(const TimeHandler &time_handler,
98 const unsigned int n_dofs_per_cell,
99 const double cell_diameter,
100 const unsigned int degree,
101 const double diffusivity,
102 const Tensor<1, dim> &convective_velocity,
103 const std::vector<Tensor<1, dim>> &grad_scalar_shape);
104} // namespace StabilizationTools

110 template <int dim>
111 inline double
112 compute_h_tau_squared(const unsigned int n_dofs_per_cell,
113 const double cell_diameter,
114 const Tensor<1, dim> &convective_velocity,
115 const double velocity_norm_squared,
116 const std::vector<Tensor<1, dim>> &grad_scalar_shape)
117 {
118 if constexpr (dim == 2)
119 {
120 // In 2D, use Tezduyar's formula to compute the squared cell length.
121 return compute_squared_cell_length_along_velocity_field(
122 n_dofs_per_cell,
123 cell_diameter,
124 convective_velocity,
125 velocity_norm_squared,
126 grad_scalar_shape);
127 }
128 else
129 {
130 // In 3D, use the cell diameter.
131 return cell_diameter * cell_diameter;
132 }
133 DEAL_II_ASSERT_UNREACHABLE();
134 }

136 template <int dim>
137 inline double compute_squared_cell_length_along_velocity_field(
138 const unsigned int n_dofs_per_cell,
139 const double cell_diameter,
140 const Tensor<1, dim> &convective_velocity,
141 const double velocity_norm_squared,
142 const std::vector<Tensor<1, dim>> &grad_scalar_shape)
143 {
144 if (velocity_norm_squared < 1e-12)
145 // Return *square* of cell_diameter
146 return cell_diameter * cell_diameter;
147
148 double sum = 0.;
149 for (unsigned int i = 0; i < n_dofs_per_cell; ++i)
150 sum += std::abs(convective_velocity * grad_scalar_shape[i]);
151 return 4. * velocity_norm_squared / (sum * sum);
152 }

154 template <int dim>
155 inline double
156 compute_tau_supg(const TimeHandler &time_handler,
157 const unsigned int n_dofs_per_cell,
158 const double cell_diameter,
159 const unsigned int degree,
160 const double diffusivity,
161 const Tensor<1, dim> &convective_velocity,
162 const std::vector<Tensor<1, dim>> &grad_scalar_shape)
163 {
164 const double velocity_norm_squared = convective_velocity.norm_square();
165 const double one_over_h_tau_squared =
166 1. / compute_h_tau_squared(n_dofs_per_cell,
167 cell_diameter,
168 convective_velocity,
169 velocity_norm_squared,
170 grad_scalar_shape);
171
172 Assert(one_over_h_tau_squared > 0, ExcInternalError());
173
174 const double conv =
175 4. * velocity_norm_squared * degree * degree * one_over_h_tau_squared;
176 double diff = 4. * diffusivity * degree * degree * one_over_h_tau_squared;
177 diff *= 9. * diff;
178
179 double tau = conv + diff;
180 if (!time_handler.is_steady())
181 {
182 const double unsteady = 1. / time_handler.get_current_timestep();
183 tau += unsteady * unsteady;
184 }
185
186 Assert(tau > 0, ExcInternalError());
187
188 tau = pow(tau, -0.5);
189 return tau;
190 }
