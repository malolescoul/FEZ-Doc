146 template <typename VectorType>
147 void reinit(const typename DoFHandler<dim>::active_cell_iterator &cell,
148 const VectorType &current_solution,
149 const std::shared_ptr<Function<dim>> &source_terms,
150 const std::shared_ptr<Function<dim>> & /*exact_solution*/)
151 {
152 fe_values.reinit(cell);
153
154 for (const unsigned int i : fe_values.dof_indices())
155 components[i] = fe_values.get_fe().system_to_component_index(i).first;
156
157 /**
158 * Volume contributions
159 */
160 fe_values[position].get_function_values(current_solution, position_values);
161 fe_values[position].get_function_gradients(current_solution,
162 present_position_gradients);
163 fe_values[position].get_function_symmetric_gradients(
164 current_solution, position_sym_gradients);
165
166 const auto &quadrature_points = fe_values.get_quadrature_points();
167 source_terms->vector_value_list(quadrature_points, source_term_full);
168
169 // First map the quadrature point to the current configuration
170 // The quadrature points in the current configuration are simply the
171 // position field interpolated at the reference space quadrature points
172 for (unsigned int q = 0; q < n_q_points; ++q)
173 qpoints_current_mesh[q] = position_values[q];
174 source_terms->vector_value_list(qpoints_current_mesh,
175 source_term_full_current_mesh);
176
177 // These are computed with finite differences by deal.II (AutoDerivative)
178 source_terms->vector_gradient_list(qpoints_current_mesh,
179 grad_source_term_full_current_mesh);
180
181 // Cahn-Hilliard moving-mesh forcing: evaluate the prescribed phase and its
182 // analytic gradient/Hessian on the current (deformed) mesh. The Hessian is
183 // needed to linearize the forcing w.r.t. the mesh position.
184 if (evaluate_chns_forcing)
185 {
186 const auto &tracer =
187 *param.initial_conditions.initial_chns_tracer_callback;
188 for (unsigned int q = 0; q < n_q_points; ++q)
189 {
190 chns_tracer_values[q] = tracer.value(qpoints_current_mesh[q]);
191 chns_tracer_gradients[q] = tracer.gradient(qpoints_current_mesh[q]);
192 chns_tracer_hessians[q] = tracer.hessian(qpoints_current_mesh[q]);
193 }
194 }
195
196 // Enlarged presolver: build the psi finite-element field on the moving
197 // (deformed) mesh, so the psi Helmholtz reconstruction is assembled on the
198 // current configuration exactly as in the full CHNS solver. The physical
199 // tracer phi stays analytic (chns_tracer_* above); only psi is an unknown.
200 if (with_enlarged_psi)
201 {
202 const MappingFEField<dim, dim, VectorType> moving_mapping(
203 cell->get_dof_handler(), current_solution, position_mask);
204 FEValues<dim> fe_values_moving(moving_mapping,
205 fe_values.get_fe(),
206 fe_values.get_quadrature(),
207 update_values | update_gradients |
208 update_quadrature_points |
209 update_JxW_values);
210 fe_values_moving.reinit(cell);
211
212 fe_values_moving[psi].get_function_values(current_solution, psi_values);
213 fe_values_moving[psi].get_function_gradients(current_solution,
214 psi_gradients);
215
216 for (unsigned int q = 0; q < n_q_points; ++q)
217 {
218 JxW_moving[q] = fe_values_moving.JxW(q);
219 for (unsigned int k = 0; k < dofs_per_cell; ++k)
220 {
221 shape_psi[q][k] = fe_values_moving[psi].value(k, q);
222 grad_shape_psi[q][k] = fe_values_moving[psi].gradient(k, q);
223 grad_phi_x_moving[q][k] = fe_values_moving[position].gradient(k, q);
224 }
225 }
226 }
227
228 for (unsigned int q = 0; q < n_q_points; ++q)
229 {
230 JxW_fixed[q] = fe_values.JxW(q);
231
232 for (unsigned int d = 0; d < dim; ++d)
233 {
234 source_term_position_fixed_mesh[q][d] = source_term_full[q](d);
235 source_term_position_current_mesh[q][d] =
236 source_term_full_current_mesh[q](d);
237
238 // Premultiply the gradient by the coefficient in front of it
239 for (unsigned int dj = 0; dj < dim; ++dj)
240 grad_source_term_position_current_mesh[q][d][dj] =
241 source_term_moving_mesh_multiplier *
242 grad_source_term_full_current_mesh[q][d][dj];
243 }
244
245 // Source term effectively used
246 // Multipliers cannot both be nonzero
247 Assert(!(std::abs(source_term_fixed_mesh_multiplier) > 1e-14 &&
248 std::abs(source_term_moving_mesh_multiplier) > 1e-14),
249 ExcInternalError());
250
251 source_term_position[q] =
252 source_term_fixed_mesh_multiplier * source_term_position_fixed_mesh[q] +
253 source_term_moving_mesh_multiplier *
254 source_term_position_current_mesh[q];
255
256 const Point<dim> &pt = quadrature_points[q];
257 lame_mu[q] =
258 param.physical_properties.pseudosolids[0].lame_mu_fun->value(pt);
259 lame_lambda[q] =
260 param.physical_properties.pseudosolids[0].lame_lambda_fun->value(pt);
261
262 AssertThrow(lame_mu[q] >= 0,
263 ExcMessage("Lamé coefficient mu should be positive"));
264
265 // Data for hyperelastic models
266 const Tensor<2, dim> &F = present_position_gradients[q];
267 present_position_J[q] = determinant(F);
268 present_position_inverse_gradients[q] = invert(F);
269 present_position_inverse_gradients_T[q] =
270 transpose(present_position_inverse_gradients[q]);
271
272 if constexpr (running_in_debug_mode())
273 {
274 // Throw if
275 }
276 // AssertThrow(present_position_J[q] > 0, ExcMessage("Inverted"));
277
278 for (unsigned int k = 0; k < dofs_per_cell; ++k)
279 {
280 phi_x[q][k] = fe_values[position].value(k, q);
281 grad_phi_x[q][k] = fe_values[position].gradient(k, q);
282 sym_grad_phi_x[q][k] = fe_values[position].symmetric_gradient(k, q);
283 div_phi_x[q][k] = fe_values[position].divergence(k, q);
284 }
285 }
286 }
