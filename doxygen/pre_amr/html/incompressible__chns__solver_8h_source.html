419 /**
420 * Gradient of source term, using finite differences
421 */
422 virtual void
423 vector_gradient(const Point<dim> &p,
424 std::vector<Tensor<1, dim>> &gradients) const override
425 {
426 const double h = 1e-8;
427
428 Vector<double> vals_plus(gradients.size()), vals_minus(gradients.size());
429
430 for (unsigned int d = 0; d < dim; ++d)
431 {
432 Point<dim> p_plus = p, p_minus = p;
433 p_plus[d] += h;
434 p_minus[d] -= h;
435
436 this->vector_value(p_plus, vals_plus);
437 this->vector_value(p_minus, vals_minus);
438
439 // Centered finite differences
440 for (unsigned int c = 0; c < gradients.size(); ++c)
441 gradients[c][d] = (vals_plus[c] - vals_minus[c]) / (2.0 * h);
442 }
443 }
