5template <int dim>
6void get_owned_mesh_vertices(const Triangulation<dim> &triangulation,
7 const types::subdomain_id subdomain_id,
8 std::vector<bool> &owned_vertices)
9{
10 owned_vertices.clear();
11 owned_vertices.resize(triangulation.n_vertices(), false);
12
13 // Start by marking all vertices touching an owned cell as owned
14 for (const auto &cell : triangulation.active_cell_iterators())
15 if (cell->is_locally_owned())
16 for (const unsigned int v : cell->vertex_indices())
17 owned_vertices[cell->vertex_index(v)] = true;
18
19 // If a ghost cell with lesser id touches a vertex, mark it non-owned
20 for (const auto &cell : triangulation.active_cell_iterators())
21 if (cell->is_artificial() ||
22 (cell->is_ghost() && cell->subdomain_id() < subdomain_id))
23 for (const unsigned int v : cell->vertex_indices())
24 owned_vertices[cell->vertex_index(v)] = false;
25}

34template <int dim>
35std::set<Point<dim>, PointComparator<dim>>
36get_mesh_vertices_on_boundary(const DoFHandler<dim> &dof_handler,
37 const types::boundary_id boundary_id)
38{
39 const MPI_Comm comm = dof_handler.get_mpi_communicator();
40 std::set<Point<dim>, PointComparator<dim>> vertices_on_boundary;
41 std::vector<Point<dim>> local_vertices_on_boundary;
42
43 // Mark based on faces
44 for (const auto &cell : dof_handler.active_cell_iterators())
45 for (const auto &face : cell->face_iterators())
46 if (face->at_boundary() && face->boundary_id() == boundary_id)
47 for (unsigned int i = 0; i < face->n_vertices(); ++i)
48 local_vertices_on_boundary.push_back(face->vertex(i));
49
50 std::vector<std::vector<Point<dim>>> gathered =
51 Utilities::MPI::all_gather(comm, local_vertices_on_boundary);
52
53 for (const auto &vec : gathered)
54 for (const auto pt : vec)
55 vertices_on_boundary.insert(pt);
56
57 return vertices_on_boundary;
58}
