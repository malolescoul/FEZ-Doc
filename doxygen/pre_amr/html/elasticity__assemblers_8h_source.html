20 /**
21 * Regularized compression coefficient of the Cahn-Hilliard moving-mesh
22 * forcing. The raw coefficient phi / (1 - gamma^2 phi^2) is singular as
23 * gamma*phi -> 1, so phi is saturated through a tanh before being used.
24 * Both the value and its derivative w.r.t. the phase are returned (the
25 * derivative is used to linearize the forcing).
26 */
27 struct MeshForcingFactor
28 {
29 double value;
30 double derivative;
31 };
32
33 inline MeshForcingFactor
34 mesh_forcing_factor(const double phase_value, const double gamma)
35 {
36 constexpr double phi_max_user = 0.998;
37 const double phi_max_safe =
38 std::min(phi_max_user, 0.98 / std::max(gamma, 1e-14));
39
40 const double z = phase_value / phi_max_safe;
41 const double tanh_z = std::tanh(z);
42 const double regularized_phase = phi_max_safe * tanh_z;
43 const double regularized_jacobian = 1. - tanh_z * tanh_z;
44 const double denominator =
45 1. - gamma * gamma * regularized_phase * regularized_phase;
46 const double support = 1. / denominator;
47 const double support_derivative = 2. * gamma * gamma * regularized_phase *
48 regularized_jacobian /
49 (denominator * denominator);
50
51 return {phase_value * support, support + phase_value * support_derivative};
52 }
53
54 /**
55 * Smooth odd approximation of sign(psi) * |psi|^q used to equalize the
56 * enlarged mesh-forcing marker before it enters the regularized factor. The
57 * small delta keeps the map differentiable at psi=0 for Newton. With q = 1
58 * the map is the identity (no-op). For 0 < q < 1 weak marker values are
59 * amplified (the mesh reacts earlier across the enlarged interface); q > 1
60 * damps them. Both the value and its derivative are returned.
61 */
62 inline MeshForcingFactor
63 smooth_power_equalized_phase(const double phase_value, const double exponent)
64 {
65 if (std::abs(exponent - 1.) < 1e-14)
66 return {phase_value, 1.};
67
68 constexpr double delta = 2e-2;
69 const double a = phase_value * phase_value + delta * delta;
70
71 return {phase_value * std::pow(a, 0.5 * (exponent - 1.)),
72 std::pow(a, 0.5 * (exponent - 3.)) *
73 (delta * delta + exponent * phase_value * phase_value)};
74 }
75
76 /**
77 * Peak of the dimensionless enlarged-compression lobe associated with the
78 * one-dimensional Helmholtz marker. Away from the much thinner physical
79 * interface, psi - L^2 psi'' = sign(x) gives
80 *
81 * L |grad psi| = 1 - |psi|.
82 *
83 * Hence the lobe shape, with s = |psi| in [0,1], is
84 *
85 * (1-s) |factor(equalize_q(s))|.
86 *
87 * A short, deterministic one-dimensional search is sufficient here. Its
88 * result is stored by the forcing assembler and reused in all FE loops.
89 */
90 inline double
91 enlarged_mesh_forcing_lobe_peak(const double gamma, const double exponent)
92 {
93 const auto lobe = [gamma, exponent](const double s) {
94 const double equalized =
95 smooth_power_equalized_phase(s, exponent).value;
96 return (1. - s) *
97 std::abs(mesh_forcing_factor(equalized, gamma).value);
98 };
99
100 // First bracket the global maximum. For positive q the lobe is smooth
101 // and unimodal; the coarse scan also keeps the search robust near very
102 // small exponents and strong regularization.
103 constexpr unsigned int n_intervals = 128;
104 unsigned int best_index = 0;
105 double best_value = 0.;
106 for (unsigned int i = 0; i <= n_intervals; ++i)
107 {
108 const double value = lobe(static_cast<double>(i) / n_intervals);
109 if (value > best_value)
110 {
111 best_value = value;
112 best_index = i;
113 }
114 }
115
116 double left = static_cast<double>(best_index == 0 ? 0 : best_index - 1) /
117 n_intervals;
118 double right =
119 static_cast<double>(std::min(n_intervals, best_index + 1)) / n_intervals;
120
121 constexpr double golden_ratio_conjugate = 0.6180339887498948482;
122 double x_left = right - golden_ratio_conjugate * (right - left);
123 double x_right = left + golden_ratio_conjugate * (right - left);
124 double value_left = lobe(x_left);
125 double value_right = lobe(x_right);
126 for (unsigned int iteration = 0; iteration < 48; ++iteration)
127 {
128 if (value_left < value_right)
129 {
130 left = x_left;
131 x_left = x_right;
132 value_left = value_right;
133 x_right = left + golden_ratio_conjugate * (right - left);
134 value_right = lobe(x_right);
135 }
136 else
137 {
138 right = x_right;
139 x_right = x_left;
140 value_right = value_left;
141 x_left = right - golden_ratio_conjugate * (right - left);
142 value_left = lobe(x_left);
143 }
144 }
145
146 return std::max(best_value, std::max(value_left, value_right));
147 }

149 /**
150 * Automatic amplitude normalization of the equalized enlarged forcing.
151 * The q=1 lobe is the reference, so changing q primarily moves/reshapes the
152 * lobes without changing their theoretical peak compression.
153 */
154 inline double
155 enlarged_mesh_forcing_normalization(const double gamma,
156 const double exponent)
157 {
158 if (std::abs(exponent - 1.) < 1e-14)
159 return 1.;
160
161 const double reference_peak =
162 enlarged_mesh_forcing_lobe_peak(gamma, 1.);
163 const double equalized_peak =
164 enlarged_mesh_forcing_lobe_peak(gamma, exponent);
165
166 return equalized_peak > 1e-14 ? reference_peak / equalized_peak : 1.;
167 }

169 /**
170 * Regularized compression coefficient for the enlarged marker psi: the
171 * equalization map is chained into mesh_forcing_factor, so the returned
172 * value is factor(equalize(psi)) and the derivative is the product of the
173 * two chain-rule terms (w.r.t. psi).
174 */
175 inline MeshForcingFactor
176 enlarged_mesh_forcing_factor(const double phase_value,
177 const double gamma,
178 const double exponent,
179 const double normalization)
180 {
181 const MeshForcingFactor equalized =
182 smooth_power_equalized_phase(phase_value, exponent);
183 MeshForcingFactor factor = mesh_forcing_factor(equalized.value, gamma);
184 factor.value *= normalization;
185 factor.derivative *= equalized.derivative * normalization;
186 return factor;
187 }

325 /**
326 * Cahn-Hilliard moving-mesh forcing source term of the pseudosolid
327 * equation, f = compression * eps * factor(phi) * grad phi. This is the
328 * "chns form" path (as opposed to a user-defined custom source term, which
329 * goes through CurrentMeshSourceAssembler).
330 *
331 * It is meant to support two ways of providing the phase marker:
332 * - field mode : phi/psi is an unknown FE field (full CHNS-ALE solver and
333 * the enlarged presolver). The Jacobian is exact and
334 * Hessian-free (material gradient remapping -G^T grad phi).
335 * - function mode: phi is an analytic function to evaluate (the ALE
336 * elasticity presolver). The Jacobian is exact thanks to
337 * the analytic Hessian of phi (symbolic differentiation).
338 * The function mode (ScratchDataElasticity) and the field mode
339 * (ScratchDataCHNS, with or without the enlarged psi marker) are both
340 * implemented. In the enlarged field mode the compression marker is the
341 * widened field psi (with its own length scale and equalized factor) while
342 * the physical compression still uses the sharp tracer phi.
343 *
344 * SIGN CONVENTION: the forcing enters the residual with rhs -= forcing (it
345 * is treated as a pseudosolid source term, matching the elasticity sign), so
346 * a POSITIVE compression factor compresses the mesh towards the interface.
347 */

380 /**
381 * Enlarged-presolver reconstruction of the widened marker psi by the
382 * Helmholtz equation
383 *
384 * psi - L^2 lap(psi) = phi_0,
385 *
386 * assembled on the moving (deformed) configuration, exactly as in the full
387 * CHNS solver (see assemble_psi_equation_* in incompressible_chns_assemblers
388 * .h). The crucial difference is that here phi_0 is an *analytic* function
389 * (the prescribed Cahn-Hilliard initial condition), not a finite-element
390 * unknown, so:
391 * - there is no psi<-phi nor psi<-mu coupling and no mu-correction;
392 * - the psi<-x block carries the extra source variation delta phi_0 =
393 * grad(phi_0) . delta x, which the field-mode CHNS assembler omits
394 * (there phi is an FE field, so its value does not transform).
395 * This is the third assembly case (hybrid: analytic phi, FE psi); it must
396 * not be confused with the field-mode CHNS solver.
397 */

427 template <int dim, typename ScratchData, typename CopyData>
428 void setup_assemblers(
429 const ParameterReader<dim> &param,
430 const ComponentOrdering &ordering,
431 std::vector<std::unique_ptr<AssemblerBase<ScratchData, CopyData>>>
432 &assemblers)
433 {
434 const auto &solid = param.physical_properties.pseudosolids[0];
435
436 switch (solid.constitutive_model)
437 {
438 case Parameters::PseudoSolid<dim>::ConstitutiveModel::linear_elasticity:
439 assemblers.emplace_back(
440 std::make_unique<
441 LinearElasticityAssembler<dim, ScratchData, CopyData>>(param,
442 ordering));
443 break;
444 case Parameters::PseudoSolid<dim>::ConstitutiveModel::neo_hookean:
445 assemblers.emplace_back(
446 std::make_unique<NeoHookeanAssembler<dim, ScratchData, CopyData>>(
447 param, ordering));
448 break;
449 case Parameters::PseudoSolid<dim>::ConstitutiveModel::ogden:
450 assemblers.emplace_back(
451 std::make_unique<
452 OgdenHyperelasticityAssembler<dim, ScratchData, CopyData>>(
453 param, ordering));
454 break;
455 default:
456 DEAL_II_ASSERT_UNREACHABLE();
457 }
458
459 // Custom (user-defined) source term on the current mesh.
460 if (param.elasticity.enable_source_term_on_current_mesh)
461 assemblers.emplace_back(
462 std::make_unique<
463 CurrentMeshSourceAssembler<dim, ScratchData, CopyData>>(param,
464 ordering));
465
466 // TODO: wire the "custom" mff source term path, where the moving-mesh
467 // forcing is a user-defined expression assembled through
468 // CurrentMeshSourceAssembler. For now only "off" and "chns form" are
469 // functional; "custom" is parsed but assembles no forcing.
470
471 // Cahn-Hilliard moving-mesh forcing ("chns form" path): function mode for
472 // the elasticity presolver (analytic phi) and field mode for the full
473 // CHNS-ALE solver (FE phi). Both share this assembler.
474 const bool with_chns_form_forcing =
475 param.cahn_hilliard.mff_source_term ==
476 Parameters::CahnHilliard<dim>::MeshForcingSourceTerm::chns_form;
477
478 if constexpr (std::is_same_v<ScratchData, ScratchDataElasticity<dim>> ||
479 std::is_same_v<
480 ScratchData,
481 NavierStokesScratch::ScratchDataCHNS<dim, true, false>> ||
482 std::is_same_v<
483 ScratchData,
484 NavierStokesScratch::ScratchDataCHNS<dim, true, true>>)
485 {
486 if (with_chns_form_forcing)
487 assemblers.emplace_back(
488 std::make_unique<
489 SourceFromCHNSTracerAssembler<dim, ScratchData, CopyData>>(
490 param, ordering));
491 }
492
493 // Enlarged presolver: reconstruct the widened marker psi by its Helmholtz
494 // equation (hybrid mode, analytic phi). Registered only when the ordering
495 // carries psi, i.e. the elasticity solver runs in enlarged-presolver mode.
496 if constexpr (std::is_same_v<ScratchData, ScratchDataElasticity<dim>>)
497 {
498 if (ordering.has_variable(SolverInfo::VariableType::phase_enlarged))
499 assemblers.emplace_back(
500 std::make_unique<
501 PresolverPsiAssembler<dim, ScratchData, CopyData>>(param,
502 ordering));
503 }
504 }
