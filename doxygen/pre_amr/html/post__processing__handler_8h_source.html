583template <int dim>
584void PostProcessingHandler<dim>::PrefixData::append_to_prefix_or_suffix(
585 const Parameters::Output &output_param,
586 const bool is_for_pvd,
587 std::string &prefix_or_suffix) const
588{
589 if (is_convergence_step)
590 prefix_or_suffix += "_convergence_step_" + std::to_string(convergence_step);
591
592 if (is_fixed_point_step)
593 if (!output_param.fixed_point.single_pvd)
594 prefix_or_suffix += "_fp_" + std::to_string(fixed_point_step);
595
596 // The interval index should not be written in the name of the pvd file
597 if (!is_for_pvd)
598 if (is_time_subinterval)
599 prefix_or_suffix += "_int_" + std::to_string(interval_index);
600
601 if (is_prerefinement_step)
602 prefix_or_suffix +=
603 "_prerefinement_step_" + std::to_string(prerefinement_step);
604}

652template <int dim>
653template <typename VectorType>
654void PostProcessingHandler<dim>::output_fields(const Mapping<dim> &mapping,
655 const VectorType &solution,
656 const TimeHandler &time_handler,
657 const PrefixData &prefix_data)
658{
659 // Get the partitions only once
660 if (subdomains.size() == 0)
661 {
662 Assert(
663 triangulation->n_active_cells() > 0,
664 ExcMessage(
665 "Cannot create subdomains vector because triangulation is empty."));
666 subdomains.reinit(triangulation->n_active_cells());
667 for (unsigned int i = 0; i < subdomains.size(); ++i)
668 subdomains(i) = triangulation->locally_owned_subdomain();
669 }
670
671 // Compute slices indices once
672 if (post_proc_param.slices.enable && slice_indices.size() == 0)
673 create_slices();
674
675 // Export fields in volume
676 if (should_output_volume_fields(time_handler))
677 output_volume_fields(mapping, solution, time_handler, prefix_data);
678
679 // Export fields on prescribed boundary (skin)
680 if (should_output_skin_fields(time_handler))
681 output_skin_fields(mapping, solution, time_handler, prefix_data);
682
683 if (mpi_rank == 0 && (should_output_volume_fields(time_handler) ||
684 should_output_skin_fields(time_handler)))
685 write_pvd(prefix_data);
686}

688template <int dim>
689template <typename VectorType>
690void PostProcessingHandler<dim>::output_volume_fields(
691 const Mapping<dim> &mapping,
692 const VectorType &solution,
693 const TimeHandler &time_handler,
694 const PrefixData &prefix_data)
695{
696 data_out->add_data_vector(solution,
697 solution_names,
698 DataOut<dim>::type_dof_data,
699 data_component_interpretation);
700 data_out->add_data_vector(subdomains, "subdomain");
701
702 // Output all the DataPostprocessors
703 for (const auto &postprocessor : postprocessors)
704 data_out->add_data_vector(solution, *postprocessor);
705
706 data_out->build_patches(mapping,
707 output_param.n_subdivisions,
708 DataOut<dim>::CurvedCellRegion::curved_inner_cells);
709
710 std::string prefix = output_param.output_prefix;
711 prefix_data.append_to_prefix_or_suffix(output_param, false, prefix);
712
713 const std::string pvtu_file =
714 data_out->write_vtu_with_pvtu_record(output_param.output_dir,
715 prefix,
716 time_handler.current_time_iteration,
717 mpi_communicator,
718 2,
719 output_param.n_vtu_groups);
720
721 if (prefix_data.is_prerefinement_step)
722 prerefinements_pseudotimes_and_names.emplace_back(
723 static_cast<double>(prefix_data.prerefinement_step), pvtu_file);
724 else

855template <int dim>
856template <typename VectorType,
857 typename MappingType,
858 typename FaceQuadratureType>
859void PostProcessingHandler<dim>::compute_forces(
860 const ComponentOrdering &ordering,
861 const DoFHandler<dim> &dof_handler,
862 const MappingType &mapping,
863 const FaceQuadratureType &face_quadrature,
864 const VectorType &solution,
865 const TimeHandler &time_handler)
866{
867 const auto &forces_param = post_proc_param.forces;
868 using Forces = Parameters::PostProcessing::Forces;
869
870 Tensor<1, dim> forces;
871 std::string method = "";
872
873 std::vector<Tensor<1, dim>> force_per_face(
874 dof_handler.get_triangulation().n_faces());
875
876 switch (forces_param.method)
877 {
878 case Forces::ComputationMethod::stress_vector:
879 {
880 method = "stress_vector";
881 const FEValuesExtractors::Vector velocity_extractor(ordering.u_lower);
882 const FEValuesExtractors::Scalar pressure_extractor(ordering.p_lower);
883
884 // FIXME: take viscosity of the mixture in CHNS
885 const double mu = physical_properties.fluids[0].dynamic_viscosity;
886
887 forces = PostProcessingTools::compute_forces_on_boundary(
888 dof_handler,
889 mapping,
890 face_quadrature,
891 solution,
892 forces_param.boundary_id,
893 velocity_extractor,
894 pressure_extractor,
895 mu,
896 force_per_face);
897 break;
898 }
899 case Forces::ComputationMethod::lagrange_multiplier:
900 {
901 method = "lagrange_multiplier";
902 AssertThrow(ordering.l_lower != numbers::invalid_unsigned_int,
903 ExcMessage(
904 "Cannot compute forces with a Lagrange multiplier "
905 "because the chosen "
906 "solver does not have a Lagrange multiplier variable."));
907
908 const FEValuesExtractors::Vector lambda_extractor(ordering.l_lower);
909 forces = PostProcessingTools::
910 compute_forces_on_boundary_with_lagrange_multiplier(
911 dof_handler,
912 mapping,
913 face_quadrature,
914 solution,
915 forces_param.boundary_id,
916 lambda_extractor,
917 force_per_face);
918 break;
919 }
920 default:
921 DEAL_II_NOT_IMPLEMENTED();
922 }
923
924 if (forces_param.verbosity == Parameters::Verbosity::verbose && mpi_rank == 0)
925 {
926 std::ios::fmtflags old_flags = std::cout.flags();
927 unsigned int old_precision = std::cout.precision();
928
929 std::vector<std::string> dim_str = {"x", "y", "z"};
930 std::cout << std::scientific << std::setprecision(forces_param.precision)
931 << std::showpos << std::endl;
932 std::cout << "Forces on boundary with id " << forces_param.boundary_id
933 << " computed with method : " << method << std::endl;
934 for (unsigned int d = 0; d < dim; ++d)
935 std::cout << "F" + dim_str[d] << " = " << forces[d] << std::endl;
936
937 std::cout.precision(old_precision);
938 std::cout.flags(old_flags);
939 }
940
941 // Add forces to forces table and write if time step matches frequency
942 {
943 add_force_to_table(forces, time_handler, forces_table);
944 if (mpi_rank == 0 && should_output_forces(time_handler))
945 {
946 std::ofstream outfile(output_param.output_dir +
947 post_proc_param.forces.output_prefix + ".txt");
948 write_table(outfile, forces_table, post_proc_param.forces);
949 }
950 }
951
952 // Compute forces on each slice of given boundary
953 const auto &slices_param = post_proc_param.slices;
954 if (slices_param.enable && slices_param.compute_forces_on_slices)
955 {
956 std::vector<Tensor<1, dim>> forces_per_slice_local(slices_param.n_slices);
957 std::vector<Tensor<1, dim>> forces_per_slice = forces_per_slice_local;
958
959 for (const auto &face : triangulation->active_face_iterators())
960 {
961 if (face->user_index() != numbers::invalid_unsigned_int)
962 forces_per_slice_local[face->user_index()] +=
963 force_per_face[face->index()];
964 }
965
966 for (unsigned int i = 0; i < slices_param.n_slices; ++i)
967 {
968 forces_per_slice[i] =
969 Utilities::MPI::sum(forces_per_slice_local[i],
970 dof_handler.get_mpi_communicator());
971 add_force_to_table(forces_per_slice[i],
972 time_handler,
973 forces_table_per_slice,
974 i);
975 }
976
977 if (forces_param.verbosity == Parameters::Verbosity::verbose &&
978 mpi_rank == 0)
979 {
980 std::ios::fmtflags old_flags = std::cout.flags();
981 unsigned int old_precision = std::cout.precision();
982
983 std::vector<std::string> dim_str = {"x", "y", "z"};
984 std::cout << std::scientific << std::setprecision(forces_param.precision)
985 << std::showpos << std::endl;
986 std::cout << "Forces per slice on boundary with id "
987 << forces_param.boundary_id << ":" << std::endl;
988 for (unsigned int i = 0; i < slices_param.n_slices; ++i)
989 {
990 std::cout << "Slice " << i << ": ";
991 for (unsigned int d = 0; d < dim; ++d)
992 std::cout << "F" + dim_str[d] << " = " << forces_per_slice[i][d]
993 << "\t";
994 std::cout << std::endl;
995 }
996
997 std::cout.precision(old_precision);
998 std::cout.flags(old_flags);
999 }
1000
1001 // Write to file
1002 if (mpi_rank == 0 &&
1003 should_output_postprocessing(time_handler, slices_param))
1004 {
1005 std::ofstream slices_outfile(output_param.output_dir +
1006 post_proc_param.forces.output_prefix + "_" +
1007 slices_param.output_prefix + ".txt");
1008 write_table(slices_outfile,
1009 forces_table_per_slice,
1010 post_proc_param.forces);
1011 }
1012
1013 // Check that sum of forces on slices is the force on boundary
1014 {
1015 Tensor<1, dim> sum_slices;
1016 for (const auto &f : forces_per_slice)
1017 sum_slices += f;
1018 AssertThrow((forces - sum_slices).norm_square() < 1e-14,
1019 ExcMessage("Sum of forces on slices does not match the total "
1020 "forces on this boundary"));
1021 }
1022 }
1023}
