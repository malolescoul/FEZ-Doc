46 // FNV-1a hash of a file's contents, used to detect mesh changes in the
47 // presolved-mesh-position cache fingerprint.
48 std::string hash_file_contents(const std::string &filename)
49 {
50 std::ifstream input(filename, std::ios::binary);
51 if (!input)
52 return "unavailable:" + filename;
53
54 std::uint64_t hash = 14695981039346656037ull;
55 char character;
56 while (input.get(character))
57 {
58 hash ^= static_cast<unsigned char>(character);
59 hash *= 1099511628211ull;
60 }
61 std::ostringstream out;
62 out << std::hex << hash;
63 return out.str();
64 }

223template <int dim>
224void ElasticitySolver<dim>::run()
225{
226 reset();
227 setup_assemblers();
228 MeshTools::read_mesh(triangulation, param);
229 setup_dofs();
230 create_zero_constraints();
231 create_nonzero_constraints();
232 create_sparsity_pattern();
233 set_initial_conditions();
234 output_results();
235
236 update_boundary_conditions();
237
238 if (param.cahn_hilliard.mff_source_term ==
239 Parameters::CahnHilliard<dim>::MeshForcingSourceTerm::chns_form)
240 {
241 /**
242 * Cahn-Hilliard moving-mesh forcing. The compression forcing is steep, so
243 * its multiplier is ramped from a small fraction up to its physical value
244 * (1) with a continuation method. The user-source multipliers are disabled.
245 */
246 scratch_data->source_term_fixed_mesh_multiplier = 0.;
247 scratch_data->source_term_moving_mesh_multiplier = 0.;
248
249 const double c_min =
250 param.elasticity.presolver_initial_compression_multiplier;
251 const unsigned int n_steps = param.elasticity.presolver_continuation_steps;
252
253 const double r =
254 n_steps > 1 ? std::pow(1.0 / c_min, 1.0 / (n_steps - 1)) : 1.;
255
256 scratch_data->chns_compression_multiplier = c_min;
257
258 for (unsigned int n = 0; n < n_steps; ++n)
259 {
260 pcout << std::endl;
261 pcout << "Continuation method - Step " << n + 1 << "/" << n_steps
262 << " : chns compression multiplier = "
263 << scratch_data->chns_compression_multiplier << std::endl;
264 pcout << std::endl;
265
266 if (param.nonlinear_solver.compare_jacobian_with_finite_differences)
267 compare_analytical_matrix_with_fd();
268 solve_nonlinear_problem(time_handler);
269
270 scratch_data->chns_compression_multiplier *= r;
271 }
272 }
273 else if (param.elasticity.enable_source_term_on_current_mesh)
274 {
275 /**
276 * Continuation method to handle possibly steep source terms evaluated
277 * on the current (deformed) mesh.
278 */
279 const double c_min =
280 param.elasticity.min_current_mesh_source_term_multiplier;
281 const double c_max =
282 param.elasticity.max_current_mesh_source_term_multiplier;
283 const unsigned int n_steps = param.elasticity.n_continuation_steps;
284
285 scratch_data->source_term_moving_mesh_multiplier = c_min;
286 scratch_data->source_term_fixed_mesh_multiplier = 0.;
287
288 // Use a geometric progression to increase the continuation parameter
289 const double r =
290 n_steps > 1 ? std::pow(c_max / c_min, 1.0 / (n_steps - 1)) : 1.;
291
292 for (unsigned int n = 0; n < n_steps; ++n)
293 {
294 pcout << std::endl;
295 pcout << "Continuation method - Step " << n + 1 << "/" << n_steps
296 << " : source term multiplier = "
297 << scratch_data->source_term_moving_mesh_multiplier << std::endl;
298 pcout << std::endl;
299
300 if (param.nonlinear_solver.compare_jacobian_with_finite_differences)
301 compare_analytical_matrix_with_fd();
302 solve_nonlinear_problem(time_handler);
303
304 scratch_data->source_term_moving_mesh_multiplier *= r;
305 }
306 }
307 else
308 {
309 // Source term is evaluated on reference mesh and problem is linear
310 // This is the case when performing a convergence study with a
311 // manufactured solution, for example.
312 scratch_data->source_term_moving_mesh_multiplier = 0.;
313 scratch_data->source_term_fixed_mesh_multiplier = 1.;
314
315 if (param.nonlinear_solver.compare_jacobian_with_finite_differences)
316 compare_analytical_matrix_with_fd();
317 solve_nonlinear_problem(time_handler);
318 }
319
320 // Write the presolved mesh position cache *before* postprocessing, which
321 // moves the mesh and would otherwise deform the support points used as cache
322 // keys.
323 if (param.elasticity.presolved_mesh_position_mode !=
324 Parameters::Elasticity::PresolvedMeshPositionMode::off)
325 write_presolved_mesh_cache();
326
327 postprocess_solution();
328}

600template <int dim>
601void ElasticitySolver<dim>::solve_linear_system()
602{
603 const auto &linear_solver_param = param.linear_solver.at(this->solver_type);
604
605 if (linear_solver_param.method ==
606 Parameters::LinearSolver::Method::direct_mumps)
607 {
608 if (linear_solver_param.reuse)
609 {
610 solve_linear_system_direct(this,
611 linear_solver_param,
612 system_matrix,
613 locally_owned_dofs,
614 zero_constraints,
615 *direct_solver_reuse);
616 }
617 else
618 solve_linear_system_direct(this,
619 linear_solver_param,
620 system_matrix,
621 locally_owned_dofs,
622 zero_constraints);
623 }
624 else if (linear_solver_param.method == Parameters::LinearSolver::Method::cg)
625 {
626 solve_linear_system_cg(this,
627 linear_solver_param,
628 system_matrix,
629 locally_owned_dofs,
630 zero_constraints);
631 }
632 else if (linear_solver_param.method ==
633 Parameters::LinearSolver::Method::gmres)
634 {
635 AssertThrow(false,
636 ExcMessage("GMRES solver is not implemented for "
637 "ElasticitySolver. Use CG unstead."));
638 }
639 else
640 {
641 AssertThrow(false, ExcMessage("No known resolution method"));
642 }

704template <int dim>
705void ElasticitySolver<dim>::write_final_msh()
706{
707 if (!param.elasticity.write_final_msh)
708 return;
709
710 AssertThrow(
711 has_msh_extension(param.mesh.filename),
712 ExcMessage("Writing the final deformed mesh requires the input mesh to "
713 "come from a Gmsh .msh file."));
714
715#if defined(DEAL_II_GMSH_WITH_API)
716 const unsigned int rank = Utilities::MPI::this_mpi_process(mpi_communicator);

927{
928 const std::string cache_file =
929 param.elasticity.presolved_mesh_position_file;
930 const std::string temporary_file = "tmp." + cache_file;
931
932 std::vector<unsigned char> dofs_to_component;
933 fill_dofs_to_component(dof_handler, locally_relevant_dofs, dofs_to_component);
934 const auto support_points =
935 DoFTools::map_dofs_to_support_points(*mapping, dof_handler);
936
937 std::vector<PresolvedMeshCacheEntry<dim>> local_entries;
938 local_entries.reserve(locally_owned_dofs.n_elements());
939 for (const auto dof : locally_owned_dofs)
940 {
941 PresolvedMeshCacheEntry<dim> entry;
942 const auto &point = support_points.at(dof);
943 for (unsigned int d = 0; d < dim; ++d)
944 entry.support_point[d] = point[d];
945 entry.component =
946 dofs_to_component[locally_relevant_dofs.index_within_set(dof)];
947 entry.value = present_solution[dof];
948 local_entries.push_back(entry);
949 }
950
951 const auto gathered =
952 Utilities::MPI::gather(mpi_communicator, local_entries, 0);
953 if (mpi_rank == 0)
954 {
955 std::vector<PresolvedMeshCacheEntry<dim>> entries;
956 for (const auto &rank_entries : gathered)
957 entries.insert(entries.end(), rank_entries.begin(), rank_entries.end());
958
959 std::ofstream cache(param.output.output_dir + temporary_file);
960 AssertThrow(cache, ExcMessage("Could not write presolved mesh cache."));
961 boost::archive::text_oarchive archive(cache);
962 const std::string fingerprint = presolved_mesh_fingerprint();
963 archive << fingerprint;
964 archive << entries;
965 }
966
967 MPI_Barrier(mpi_communicator);
968 replace_temporary_files(param.output.output_dir,
969 temporary_file,
970 cache_file,
971 mpi_communicator);
972 pcout << "Wrote presolved mesh position cache to "
973 << param.output.output_dir + cache_file << std::endl;

976template <int dim>
977bool ElasticitySolver<dim>::try_load_presolved_mesh_cache()
978{
979 using Mode = Parameters::Elasticity::PresolvedMeshPositionMode;
980 const auto mode = param.elasticity.presolved_mesh_position_mode;
981 AssertThrow(mode == Mode::reuse,
982 ExcMessage("try_load_presolved_mesh_cache should only be called "
983 "in 'reuse' mode."));
984
985 reset();
986 MeshTools::read_mesh(triangulation, param);
987 setup_dofs();
988
989 const std::string cache_file =
990 param.output.output_dir + param.elasticity.presolved_mesh_position_file;
991 const std::string expected_fingerprint = presolved_mesh_fingerprint();
992
993 bool usable = true;
994 std::string reason;
995 std::vector<PresolvedMeshCacheEntry<dim>> entries;
996 {
997 std::ifstream cache(cache_file);
998 if (!cache)
999 {
1000 usable = false;
1001 reason = "missing cache file " + cache_file;
1002 }
1003 else
1004 try
1005 {
1006 boost::archive::text_iarchive archive(cache);
1007 std::string cached_fingerprint;
1008 archive >> cached_fingerprint;
1009 if (cached_fingerprint != expected_fingerprint)
1010 {
1011 usable = false;
1012 reason = "presolver-defining parameters changed";
1013 }
1014 else
1015 archive >> entries;
1016 }
1017 catch (const std::exception &exception)
1018 {
1019 usable = false;
1020 reason = "could not read cache: " + std::string(exception.what());
1021 }
1022 }
1023
1024 const auto fail = [&](const std::string &message) -> bool {
1025 pcout << "Presolved mesh position cache cannot be reused: " << message
1026 << std::endl;
1027 return false;
1028 };
1029
1030 if (Utilities::MPI::min(usable ? 1 : 0, mpi_communicator) != 1)
1031 return fail(reason.empty() ? "cache invalid on another MPI rank" : reason);
1032
1033 std::map<std::string, double> cached_values;
1034 for (const auto &entry : entries)
1035 cached_values[cache_entry_key<dim>(entry.support_point, entry.component)] =
1036 entry.value;
1037
1038 std::vector<unsigned char> dofs_to_component;
1039 fill_dofs_to_component(dof_handler, locally_relevant_dofs, dofs_to_component);
1040 const auto support_points =
1041 DoFTools::map_dofs_to_support_points(*mapping, dof_handler);
1042
1043 local_evaluation_point = 0.;
1044 bool found = true;
1045 std::string missing;
1046 for (const auto dof : locally_owned_dofs)
1047 {
1048 const unsigned int component =
1049 dofs_to_component[locally_relevant_dofs.index_within_set(dof)];
1050 std::array<double, dim> point;
1051 for (unsigned int d = 0; d < dim; ++d)
1052 point[d] = support_points.at(dof)[d];
1053 const auto value =
1054 cached_values.find(cache_entry_key<dim>(point, component));
1055 if (value == cached_values.end())
1056 {
1057 found = false;
1058 missing = "cache is missing local support points";
1059 break;
1060 }
1061 local_evaluation_point[dof] = value->second;
1062 }
1063
1064 if (Utilities::MPI::min(found ? 1 : 0, mpi_communicator) != 1)
1065 return fail(missing.empty() ? "support points missing on another rank" :
1066 missing);
1067
1068 local_evaluation_point.compress(VectorOperation::insert);
1069 present_solution = local_evaluation_point;
1070 present_solution.update_ghost_values();
1071 evaluation_point = present_solution;
1072
1073 pcout << "Loaded presolved mesh position cache from " << cache_file
1074 << std::endl;
1075 return true;
1076}
