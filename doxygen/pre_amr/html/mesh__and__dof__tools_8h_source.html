38/**
39 * Copy selected components of a finite-element solution from a source
40 * DoFHandler to a destination DoFHandler defined on the *same* triangulation
41 * (hence the same parallel partition), possibly with different component
42 * layouts. @p source_comp_to_dest_comp maps source components to destination
43 * components; mapped components must share their unit support points.
44 *
45 * Only locally owned destination cells are written. Because both DoFHandlers
46 * share the triangulation, a locally owned destination cell corresponds to a
47 * locally owned source cell, so the source values are always available (no
48 * ghost exchange is required). The caller must compress @p destination after
49 * the call.
50 */
51template <int dim, typename VectorType>
52void extract_subsolution(
53 const DoFHandler<dim> &dof_handler_source,
54 const DoFHandler<dim> &dof_handler_destination,
55 const VectorType &source,
56 VectorType &destination,
57 const std::map<unsigned int, unsigned int> &source_comp_to_dest_comp)
58{
59 const auto &fe_source = dof_handler_source.get_fe();
60 const auto &fe_destination = dof_handler_destination.get_fe();
61
62 // Precompute, once, the source-dof -> destination-local-dof correspondence
63 // (identical for every cell of a non-hp discretization).
64 std::vector<unsigned int> source_to_destination(
65 fe_source.dofs_per_cell, numbers::invalid_unsigned_int);
66
67 const auto &source_support_points = fe_source.get_unit_support_points();
68 const auto &destination_support_points =
69 fe_destination.get_unit_support_points();
70
71 for (unsigned int i = 0; i < fe_source.dofs_per_cell; ++i)
72 {
73 const unsigned int source_component =
74 fe_source.system_to_component_index(i).first;
75 const auto mapped_component =
76 source_comp_to_dest_comp.find(source_component);
77 if (mapped_component == source_comp_to_dest_comp.end())
78 continue;
79
80 for (unsigned int j = 0; j < fe_destination.dofs_per_cell; ++j)
81 if (fe_destination.system_to_component_index(j).first ==
82 mapped_component->second &&
83 source_support_points[i].distance(destination_support_points[j]) <
84 1e-12)
85 {
86 source_to_destination[i] = j;
87 break;
88 }
89
90 AssertThrow(source_to_destination[i] != numbers::invalid_unsigned_int,
91 ExcMessage("Could not match a source support point in the "
92 "destination finite-element system."));
93 }
94
95 auto source_cell = dof_handler_source.begin_active();
96 auto destination_cell = dof_handler_destination.begin_active();
97
98 Vector<double> source_local_values(fe_source.dofs_per_cell);
99 std::vector<types::global_dof_index> destination_local_indices(
100 fe_destination.dofs_per_cell);
101
102 for (; source_cell != dof_handler_source.end();
103 ++source_cell, ++destination_cell)
104 {
105 AssertThrow(destination_cell != dof_handler_destination.end(),
106 ExcMessage("Source and destination triangulations have "
107 "different active-cell layouts."));
108 if (!destination_cell->is_locally_owned())
109 continue;
110
111 source_cell->get_dof_values(source, source_local_values);
112 destination_cell->get_dof_indices(destination_local_indices);
113
114 for (unsigned int i = 0; i < fe_source.dofs_per_cell; ++i)
115 if (source_to_destination[i] != numbers::invalid_unsigned_int)
116 destination[destination_local_indices[source_to_destination[i]]] =
117 source_local_values[i];
118 }
119
120 AssertThrow(destination_cell == dof_handler_destination.end(),
121 ExcMessage("Source and destination triangulations have "
122 "different numbers of active cells."));
123}
