8 * Get the update flags for the FEValues, depending on the enabled features.

10 template <unsigned int update_flags>
11 UpdateFlags get_cell_update_flags(const bool enable_stabilization)
12 {
13 // Flags for Navier-Stokes on fixed mesh only
14 UpdateFlags flags = update_values | update_gradients |
15 update_quadrature_points | update_JxW_values;
16
17 // Pseudo-solid: also update full Jacobian matrix
18 if constexpr ((update_flags & pseudo_solid) != 0)
19 flags |= update_jacobians;
20
21 // SUPG/PSPG stabilization: also update hessians to compute strong residuals
22 if (enable_stabilization)
23 flags |= update_hessians; // | update_inverse_jacobians;
24
25 return flags;
26 }

389 template <int dim, unsigned int update_flags>
390 void ScratchData<dim, update_flags>::initialize_cahn_hilliard()
391 {
392 AssertThrow(
393 ordering.phi_lower != numbers::invalid_unsigned_int &&
394 ordering.mu_lower != numbers::invalid_unsigned_int,
395 ExcMessage(
396 "Cannot create ScratchData with Cahn Hilliard data because solver does "
397 "not have a tracer and/or potential variable(s)."));
398
399 tracer.component = phi_lower = ordering.phi_lower;
400 potential.component = mu_lower = ordering.mu_lower;
401
402 if constexpr (enable_enlarged)
403 {
404 AssertThrow(
405 ordering.psi_lower != numbers::invalid_unsigned_int,
406 ExcMessage("Cannot create enlarged ScratchData because the solver does "
407 "not have an enlarged tracer (psi) variable."));
408 psi.component = psi_lower = ordering.psi_lower;
409 }
410
411 density0 = physical_properties.fluids[0].density;
412 density1 = physical_properties.fluids[1].density;
413 const double nu0 = physical_properties.fluids[0].kinematic_viscosity;
414 const double nu1 = physical_properties.fluids[1].kinematic_viscosity;
415 dynamic_viscosity0 = density0 * nu0;
416 dynamic_viscosity1 = density1 * nu1;
417 mobility = cahn_hilliard_param.mobility;
418 epsilon = cahn_hilliard_param.epsilon_interface;
419 sigma_tilde = 3. / (2. * sqrt(2.)) * cahn_hilliard_param.surface_tension;
420 adaptive_mobility_coefficient = 0.;
421 adaptive_mobility_delta = 0.;
422 if (CahnHilliard::is_adaptive_mobility_model(cahn_hilliard_param))
423 {
424 const auto scaling =
425 CahnHilliard::get_adaptive_mobility_scaling(cahn_hilliard_param);
426 adaptive_mobility_coefficient = scaling.coefficient;
427 adaptive_mobility_delta = scaling.delta;
428 }
429 body_force = physical_properties.body_force;
430 tracer_limiter = CahnHilliard::get_limiter_function(cahn_hilliard_param);
431
432 // Select the mobility evaluator once. It returns M, its tracer derivatives,
433 // and the adaptative-mobility sensitivity at each quadrature point; this
434 // keeps model selection outside the reinit hot loop.
435 mobility_evaluation_function =
436 CahnHilliard::get_mobility_evaluation_function(cahn_hilliard_param);
437 mobility_tracer_limiter =
438 CahnHilliard::get_mobility_limiter_function(cahn_hilliard_param);
439
440 // Material marker m(phi) = q (abels_nlm) or phi (else) with its first two
441 // derivatives; selected once here so the hot loop carries no model branch.
442 material_phase_function =
443 CahnHilliard::get_material_phase_function(cahn_hilliard_param);
444 material_phase_derivative_function =
445 CahnHilliard::get_material_phase_derivative_function(cahn_hilliard_param);
446 material_phase_second_derivative_function =
447 CahnHilliard::get_material_phase_second_derivative_function(
448 cahn_hilliard_param);
449
450 if constexpr (enable_enlarged)
451 {
452 // Computed after epsilon is set above.
453 const double psi_length_scale =
454 cahn_hilliard_param.psi_interface_width_factor * epsilon;
455 psi_length_scale_sq = psi_length_scale * psi_length_scale;
456 psi_mu_correction_factor = cahn_hilliard_param.psi_mu_correction_factor;
457 }
