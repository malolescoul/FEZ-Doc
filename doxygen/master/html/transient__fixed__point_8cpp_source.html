46template <int dim>
47void TransientFixedPointData<dim>::reinit(
48 const unsigned int new_n_time_intervals,
49 parallel::DistributedTriangulationBase<dim> *&triangulation,
50 DoFHandler<dim> *&dof_handler,
51 LA::ParVectorType *&present_solution,
52 std::vector<LA::ParVectorType> *&solver_previous_solutions,
53 MetricField<dim> *&metric_for_adaptation)
54{
55 // Clear and reallocate for the new number of intervals
56 if (triangulations.size() > 0)
57 clear();
58
59 n_time_intervals = new_n_time_intervals;
60
61 triangulations.resize(n_time_intervals);
62 dof_handlers.resize(n_time_intervals);
63 present_solutions.resize(n_time_intervals);
64 previous_solutions.resize(n_time_intervals);
65 metrics_for_adaptation.resize(n_time_intervals);
66
67 for (unsigned int i = 0; i < n_time_intervals; ++i)
68 {
69 // Use a parallel::distributed::Triangulation if tree-based AMR is enabled.
70 // Otherwise, use a parallel::fullydistributed::Triangulation.
71 if (param.mesh.adaptation.strategy ==
72 Parameters::Mesh::Adaptation::Strategy::LocalRefinement)
73#if defined(DEAL_II_WITH_P4EST)
74 triangulations[i] =
75 std::make_unique<parallel::distributed::Triangulation<dim>>(
76 mpi_communicator,
77 Triangulation<dim>::limit_level_difference_at_vertices);
78#else
79 AssertThrow(false, ExcDealiiNeedsP4EST());
80#endif
81 else
82 triangulations[i] =
83 std::make_unique<parallel::fullydistributed::Triangulation<dim>>(
84 mpi_communicator);
85
86 dof_handlers[i] = std::make_unique<DoFHandler<dim>>(*triangulations[i]);
87 present_solutions[i] = std::make_unique<LA::ParVectorType>();
88 previous_solutions[i] = std::make_unique<std::vector<LA::ParVectorType>>();
89 metrics_for_adaptation[i] = std::make_unique<MetricField<dim>>();
90 }
91
92 // Assign the data at interval 0 to the passed non-owning pointers
93 this->set_interval_data(0,
94 triangulation,
95 dof_handler,
96 present_solution,
97 solver_previous_solutions,
98 metric_for_adaptation);
99}

246template <int dim>
247template <int n_components>
248void TransientFixedPointData<dim>::do_solution_transfer(
249 const unsigned int interval_index,
250 const Mapping<dim> &mapping,
251 Function<dim> &exact_solution,
252 const TimeHandler &time_handler,
253 const IndexSet &locally_relevant_dofs,
254 const std::vector<unsigned char> &dofs_to_component)
255{
256 if (param.mesh.adaptation.metric.transfer_solution)
257 {
258 /**
259 * Transfer solution from the previous interval in parallel, using the
260 * Utilities::MPI::RemotePointEvaluation tools.
261 *
262 * This is (a priori) the worst possible case for solution transfer, as we
263 * use different fully_distributed triangulations across intervals, whose
264 * partitions are completely unrelated.
265 *
266 * This function is not expected to scale well, but it is only called
267 * N_intervals - 1 times, where N_intervals is typically small compared to
268 * the number of time steps per interval.
269 */
270
271 // Data on the current interval, to be transferred into
272 const auto &dh = *dof_handlers[interval_index];
273 auto &solution = *present_solutions[interval_index];
274 auto &previous_solutions_current = *previous_solutions[interval_index];
275
276 // Data on the previous interval, to be transferred from
277 const auto &dh_prev = *dof_handlers[interval_index - 1];
278 const auto &solution_prev = *present_solutions[interval_index - 1];
279 const auto &previous_solutions_prev =
280 *previous_solutions[interval_index - 1];
281
282 Assert(previous_solutions_current.size() == previous_solutions_prev.size(),
283 ExcInternalError());
284
285 LA::ParVectorType distributed_solution(dh.locally_owned_dofs(),
286 mpi_communicator);
287
288 /**
289 * The VectorTools::point_values() function takes a vector of evaluation
290 * points and returns the interpolated values in the same order, then we
291 * need to assign them in the solution vector.
292 *
293 * Here we create the vectors of evaluation points and of dofs at these
294 * points for each component from the map of (locally relevant) support
295 * points.
296 *
297 * For FESystems with more than one component, *all* components will be
298 * interpolated at *all* evaluation points, but not all these components
299 * have corresponding dofs (e.g., with a P2-P1 Taylor-Hood element, pressure
300 * dofs will be interpolated at P2 dofs support points even if there is no
301 * pressure dof there). Set the dofs to invalid by default for all
302 * components, then overwrite and only transfer in the solution vector if
303 * the dof is valid for that component.
304 */
305
306 // The (locally relevant) support points at which the solutions on the
307 // previous interval are evaluated.
308 const std::map<types::global_dof_index, Point<dim>> support_points =
309 DoFTools::map_dofs_to_support_points(mapping, dh);
310
311 // Convert into a map [support_point : dofs]
312 std::map<Point<dim>,
313 std::vector<types::global_dof_index>,
314 PointComparator<dim>>
315 points_to_dofs;
316 for (const auto &[dof, pt] : support_points)
317 points_to_dofs[pt].push_back(dof);
318
319 const unsigned int n_pts = points_to_dofs.size();
320
321 // Convert map into vector
322 std::vector<std::vector<types::global_dof_index>> local_dofs(
323 n_pts,
324 std::vector<types::global_dof_index>(n_components,
325 numbers::invalid_unsigned_int));
326 std::vector<Point<dim>> evaluation_points(n_pts);
327
328 unsigned int i = 0;
329 for (const auto &[pt, dofs] : points_to_dofs)
330 {
331 evaluation_points[i] = pt;
332
333 for (const auto dof : dofs)
334 {
335 const unsigned char comp =
336 dofs_to_component[locally_relevant_dofs.index_within_set(dof)];
337
338 Assert(comp != static_cast<unsigned char>(-1), ExcInternalError());
339 Assert(local_dofs[i][comp] == numbers::invalid_unsigned_int,
340 ExcInternalError());
341
342 local_dofs[i][comp] = dof;
343 }
344 ++i;
345 }
346
347 // This guy then does the heavy-lifting
348 Utilities::MPI::RemotePointEvaluation<dim, dim> cache;
349
350 // Transfer current solution from the previous interval onto this interval.
351 // Call point_values() overload with 5 arguments to initialize the cache.
352 // The subsequent calls use 3 arguments, which does not reinitialize it.
353 // See also the comments in vector_tools_evaluate.h in deal.II.
354 const auto transferred_solution = VectorTools::point_values<n_components>(
355 mapping, dh_prev, solution_prev, evaluation_points, cache);
356
357 Assert(cache.all_points_found(), ExcInternalError());
358
359 for (unsigned int i = 0; i < n_pts; ++i)
360 {
361 // For 1 component, transferred_solution is a vector of Tensor<1,0>,
362 // thus a vector of double which are not subscriptable.
363 if constexpr (n_components == 1)
364 distributed_solution[local_dofs[i][0]] = transferred_solution[i];
365 else
366 for (unsigned int c = 0; c < n_components; ++c)
367 if (local_dofs[i][c] != numbers::invalid_unsigned_int)
368 distributed_solution[local_dofs[i][c]] = transferred_solution[i][c];
369 }
370 distributed_solution.compress(VectorOperation::insert);
371 solution = distributed_solution;
372
373 // Then transfer the previous solutions defined on the previous interval.
374 // Call point_values() overload with 3 arguments, which does *not*
375 // reinitialize the cache.
376 for (unsigned int k = 0; k < previous_solutions_current.size(); ++k)
377 {
378 const auto &previous_sol_prev = previous_solutions_prev[k];
379 const auto transferred_solution =
380 VectorTools::point_values<n_components>(cache,
381 dh_prev,
382 previous_sol_prev);
383
384 Assert(cache.all_points_found(), ExcInternalError());
385
386 for (unsigned int i = 0; i < n_pts; ++i)
387 {
388 if constexpr (n_components == 1)
389 distributed_solution[local_dofs[i][0]] = transferred_solution[i];
390 else
391 for (unsigned int c = 0; c < n_components; ++c)
392 if (local_dofs[i][c] != numbers::invalid_unsigned_int)
393 distributed_solution[local_dofs[i][c]] =
394 transferred_solution[i][c];
395 }
396 distributed_solution.compress(VectorOperation::insert);
397 previous_solutions_current[k] = distributed_solution;
398 }
399 }
400 else
401 {
402 // Interpolate the exact solutions at the beginning of this interval.
403 auto &dof_handler = *dof_handlers[interval_index];
404 auto &present_solution = *present_solutions[interval_index];
405 auto &this_previous_solutions = *previous_solutions[interval_index];
406
407 LA::ParVectorType distributed_solution(dof_handler.locally_owned_dofs(),
408 mpi_communicator);
409
410 // Interpolate present solution
411 VectorTools::interpolate(mapping,
412 dof_handler,
413 exact_solution,
414 distributed_solution);
415 present_solution = distributed_solution;
416
417 // Interpolate previous solutions, at previous times
418 for (unsigned int k = 0; k < this_previous_solutions.size(); ++k)
419 {
420 exact_solution.set_time(time_handler.simulation_times[1 + k]);
421 VectorTools::interpolate(mapping,
422 dof_handler,
423 exact_solution,
424 distributed_solution);
425 this_previous_solutions[k] = distributed_solution;
426 }
427
428 // Restore time
429 exact_solution.set_time(time_handler.simulation_times[0]);
430 }
431}

433template <int dim>
434void TransientFixedPointData<dim>::transfer_solution_between_intervals(
435 const unsigned int interval_index,
436 const Mapping<dim> &mapping,
437 Function<dim> &exact_solution,
438 const TimeHandler &time_handler,
439 const IndexSet &locally_relevant_dofs,
440 const std::vector<unsigned char> &dofs_to_component)
441{
442 TimerOutput::Scope t(timer, "Transfer solutions between intervals");
443
444 AssertIndexRange(interval_index, n_time_intervals);
445 Assert(interval_index > 0, ExcInternalError());
446 Assert(dofs_to_component.size() > 0,
447 ExcMessage(
448 "Assign each dof to its FE component before calling this function"));
449
450 const unsigned int n_components =
451 dof_handlers[interval_index]->get_fe().n_components();
452
453 /**
454 * VectorTools::point_values<n_components> needs the number of components
455 * at compile time, which we currently do not have access to from the NS
456 * base class.
457 *
458 * FIXME: would be cleaner to have a way to get it, but that probably
459 * involves adding a template parameter to the NS base class...
460 */
461 switch (n_components)
462 {
463 case 1:
464 do_solution_transfer<1>(interval_index,
465 mapping,
466 exact_solution,
467 time_handler,
468 locally_relevant_dofs,
469 dofs_to_component);
470 break;
471 case 2:
472 do_solution_transfer<2>(interval_index,
473 mapping,
474 exact_solution,
475 time_handler,
476 locally_relevant_dofs,
477 dofs_to_component);
478 break;
479 case 3:
480 do_solution_transfer<3>(interval_index,
481 mapping,
482 exact_solution,
483 time_handler,
484 locally_relevant_dofs,
485 dofs_to_component);
486 break;
487 case 4:
488 do_solution_transfer<4>(interval_index,
489 mapping,
490 exact_solution,
491 time_handler,
492 locally_relevant_dofs,
493 dofs_to_component);
494 break;
495 case 5:
496 do_solution_transfer<5>(interval_index,
497 mapping,
498 exact_solution,
499 time_handler,
500 locally_relevant_dofs,
501 dofs_to_component);
502 break;
503 case 6:
504 do_solution_transfer<6>(interval_index,
505 mapping,
506 exact_solution,
507 time_handler,
508 locally_relevant_dofs,
509 dofs_to_component);
510 break;
511 case 7:
512 do_solution_transfer<7>(interval_index,
513 mapping,
514 exact_solution,
515 time_handler,
516 locally_relevant_dofs,
517 dofs_to_component);
518 break;
519 case 8:
520 do_solution_transfer<8>(interval_index,
521 mapping,
522 exact_solution,
523 time_handler,
524 locally_relevant_dofs,
525 dofs_to_component);
526 break;
527 case 9:
528 do_solution_transfer<9>(interval_index,
529 mapping,
530 exact_solution,
531 time_handler,
532 locally_relevant_dofs,
533 dofs_to_component);
534 break;
535 case 10:
536 do_solution_transfer<10>(interval_index,
537 mapping,
538 exact_solution,
539 time_handler,
540 locally_relevant_dofs,
541 dofs_to_component);
542 break;
543 default:
544 // If not enough, simply add the case when needed (or a better solution
545 // will have been found by then (-: )
546 AssertThrow(
547 false,
548 ExcMessage(
549 "Solution transfer is not implemented for this number of FESystem "
550 "components (" +
551 std::to_string(n_components) +
552 "). This is very easy to solve however, as you only need to add a "
553 "call to do_solution_transfer with this number of components."));
554 }

557template <int dim>
558void TransientFixedPointData<dim>::transfer_solution_between_refinements(
559 const IndexSet &locally_relevant_dofs,
560 const AffineConstraints<double> &nonzero_constraints)
561{
562 TimerOutput::Scope t(timer, "Transfer solution between refinements");
563
564 // Adapt only the first interval for now
565 const auto &dh = *dof_handlers[0];
566 auto &present_solution = *present_solutions[0];
567 auto &previous_sols = *previous_solutions[0];
568
569 LA::ParVectorType completely_distributed_solution(dh.locally_owned_dofs(),
570 mpi_communicator);
571
572 std::vector<LA::ParVectorType> all_out(previous_sols.size() + 1);
573 for (unsigned int i = 0; i < all_out.size(); ++i)
574 all_out[i] = completely_distributed_solution;
575
576 Assert(solution_transfer, ExcInternalError());
577
578 // Interpolate the (ghosted) solutions stored in the SolutionTransfer object.
579 solution_transfer->interpolate(all_out);
580
581 // Apply the nonhomogeneous and hanging nodes constraints to the current sol
582 nonzero_constraints.distribute(all_out[0]);
583 present_solution = all_out[0];
584
585 /**
586 * At this point, we only have the nonzero_constraints for the current time
587 * available, but not for the previous times, see also the discussion in
588 * step-86.cc (transfer_solution_vectors_to_new_mesh).
589 *
590 * What we can do, however, is create the hanging node constraints for the
591 * current mesh, and apply them to the previous solutions.
592 */
593 AffineConstraints<double> hanging_node_constraints(dh.locally_owned_dofs(),
594 locally_relevant_dofs);
595 DoFTools::make_hanging_node_constraints(dh, hanging_node_constraints);
596 hanging_node_constraints.close();
597
598 for (unsigned int i = 0; i < previous_sols.size(); ++i)
599 {
600 hanging_node_constraints.distribute(all_out[i + 1]);
601 previous_sols[i] = all_out[i + 1];
602 }
603}

605template <int dim>
606void TransientFixedPointData<dim>::scale_metrics(
607 const unsigned int metric_index,
608 const TimeHandler &time_handler)
609{
610 if (param.time_integration.is_steady())
611 {
612 // Scaling for steady-state adaptation.
613 AssertThrow(n_time_intervals == 1,
614 ExcMessage(
615 "When solving for steady-state solution, a single time "
616 "subinterval is expected."));
617 metrics_for_adaptation[0]->apply_optimal_steady_multiscale_scaling();
618 }
619 else
620 {
621 // Scaling for the transient method.
622
623 // Target average spatial complexity on each time interval.
624 // If this is a convergence study with anisotropic adaptation, get this
625 // value from the MMS parameters.
626 const double Navg =
627 param.mms_param.enable ?
628 (double)param.mms_param.n_target_vertices :
629 (double)param.metric_fields[metric_index].multiscale.n_target_vertices;
630
631 const auto &n_steps_on_each_interval =
632 time_handler.n_steps_on_each_interval;
633
634 AssertDimension(n_steps_on_each_interval.size(), n_time_intervals);
635
636 // Space-time complexity N_st.
637 // If we want Navg vertices for each mesh on which "n_steps" time steps
638 // were performed, the total space-time complexity should be the sum
639 // of Navg * n_steps over each interval.
640
641 // FIXME: quadruple-check this during the convergence study
642 // const double Nst = Navg * n_time_intervals;
643
644 double Nst = 0;
645 for (auto n_steps : n_steps_on_each_interval)
646 {
647 Assert(n_steps > 0, ExcInternalError());
648 Nst += Navg * n_steps;
649 }
650
651 // FIXME: use general exponents for higher order solutions
652 const double p = (double)param.metric_fields[metric_index].multiscale.p;
653 const double d = (double)dim;
654 const double den = 2. * p + d;
655 const double exponent = p / den;
656 const double exponent_int_steps = 2. * p / den;
657 const double exponent_local_scaling = -1. / den;
658 const double exponent_local_scaling_dt = -2. / den;
659
660 // Compute the global scaling factors from the collection of metrics
661 double global_scaling = 0.;
662 for (unsigned int i = 0; i < n_time_intervals; ++i)
663 {
664 auto &metric = *metrics_for_adaptation[i];
665
666 // Update the FE solution to compute integral of determinant
667 metric.metrics_to_tensor_solution();
668 const double Ki = metric.compute_integral_determinant(exponent);
669 global_scaling +=
670 Ki * std::pow(n_steps_on_each_interval[i], exponent_int_steps);
671
672 // Apply local scaling by (det Q) ^ (-1 / (2 * p + d)) * (int_interval
673 // (dt)^-1) ^(- 2 / (2 * p + d))
674 for (auto &m : metric.metrics)
675 m *= std::pow(determinant(m), exponent_local_scaling) *
676 std::pow(n_steps_on_each_interval[i], exponent_local_scaling_dt);
677 }
678
679 // Apply global scaling (operator *= includes update of FE solution and
680 // ghosts)
681 for (const auto &m_ptr : metrics_for_adaptation)
682 {
683 (*m_ptr) *= std::pow(Nst / global_scaling, 2. / d);
684 m_ptr->is_scaled = true;
685 }
686 }
687}

696template <int dim>
697void TransientFixedPointData<dim>::adapt_meshes(const Vector<float> &criteria)
698{
699 if (!param.mesh.adaptation.enable)
700 return;
701
702 switch (param.mesh.adaptation.strategy)
703 {
704 case Parameters::Mesh::Adaptation::Strategy::RiemannianMetric:
705 {
706 /**
707 * For each time interval, adapt the associated mesh with mmg, using
708 * the designated Riemannian metric.
709 */
710 this->adapt_meshes_with_mmg();
711 break;
712 }
713 case Parameters::Mesh::Adaptation::Strategy::LocalRefinement:
714 {
715 /**
716 * Adapt the triangulation with the routines from deal.II.
717 */
718 this->adapt_mesh_with_dealii_routines(criteria);
719 break;
720 }
721 default:
722 DEAL_II_ASSERT_UNREACHABLE();
723 }
724}

732
733 if (verbose && !param.time_integration.is_steady())

763template <int dim>
764void TransientFixedPointData<dim>::adapt_mesh_with_dealii_routines(
765 const Vector<float> &criteria)
766{
767 const bool verbose =
768 Utilities::MPI::this_mpi_process(mpi_communicator) == 0 &&
769 param.mesh.adaptation.verbosity == Parameters::Verbosity::verbose;
770
771 if (verbose)
772 {
773 std::cout << std::endl;
774 std::cout << "-- Adapting tree-based mesh" << std::endl;
775 }
776
777#if defined(DEAL_II_WITH_P4EST)
778
779 // Adapt only the first interval for now
780 auto &triangulation = *triangulations[0];
781 const auto &dh = *dof_handlers[0];
782 const auto &present_solution = *present_solutions[0];
783 const auto &previous_sols = *previous_solutions[0];
784
785 const auto &tree_amr = param.mesh.adaptation.tree_amr;
786
787 // Mark cells for refinement and coarsening
788 switch (tree_amr.refinement_strategy)
789 {
790 case Parameters::Mesh::Adaptation::TreeAMR::RefinementStrategy::FixedNumber:
791 parallel::distributed::GridRefinement::refine_and_coarsen_fixed_number(
792 triangulation,
793 criteria,
794 tree_amr.fraction_to_refine,
795 tree_amr.fraction_to_coarsen,
796 tree_amr.max_n_cells);
797 break;
798 case Parameters::Mesh::Adaptation::TreeAMR::RefinementStrategy::
799 FixedFraction:
800 parallel::distributed::GridRefinement::refine_and_coarsen_fixed_fraction(
801 triangulation,
802 criteria,
803 tree_amr.fraction_to_refine,
804 tree_amr.fraction_to_coarsen);
805 break;
806 default:
807 DEAL_II_ASSERT_UNREACHABLE();
808 }
809
810 // Bound refinement according to prescribed min and max allowed levels
811 if (triangulation.n_levels() > tree_amr.max_level)
812 for (const auto &cell :
813 triangulation.active_cell_iterators_on_level(tree_amr.max_level))
814 cell->clear_refine_flag();
815 for (const auto &cell :
816 triangulation.active_cell_iterators_on_level(tree_amr.min_level))
817 cell->clear_coarsen_flag();
818
819 // (Re-)initialize the solution transfer object, and adapt the mesh
820 solution_transfer =
821 std::make_unique<SolutionTransfer<dim, LA::ParVectorType>>(dh);
822
823 // Create the vector of pointers to give the solution_transfer.
824 // Each entry must point to a *ghosted* parallel vector.
825 std::vector<const LA::ParVectorType *> all_in(previous_sols.size() + 1);
826 all_in[0] = &present_solution;
827 for (unsigned int i = 0; i < previous_sols.size(); ++i)
828 all_in[i + 1] = &previous_sols[i];
829
830 for (auto vec : all_in)
831 {
832 Assert(vec->size() == dh.n_dofs(), ExcInternalError());
833 Assert(vec->has_ghost_elements(), ExcInternalError());
834 }
835
836 triangulation.prepare_coarsening_and_refinement();
837 solution_transfer->prepare_for_coarsening_and_refinement(all_in);
838
839 // Now actually refine the mesh
840 triangulation.execute_coarsening_and_refinement();
841
842 if (verbose)
843 {
844 std::cout << "-- \tNumber of cells after adaptation: "
845 << triangulation.n_global_active_cells() << std::endl;
846 }
847
848#else
849 AssertThrow(false, ExcDealiiNeedsP4EST());
850#endif
851}
