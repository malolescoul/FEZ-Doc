393 /**
394 * Gradient of source term, using finite differences
395 */
396 virtual void
397 vector_gradient(const Point<dim> &p,
398 std::vector<Tensor<1, dim>> &gradients) const override
399 {
400 const double h = 1e-8;
401
402 Vector<double> vals_plus(gradients.size()), vals_minus(gradients.size());
403
404 for (unsigned int d = 0; d < dim; ++d)
405 {
406 Point<dim> p_plus = p, p_minus = p;
407 p_plus[d] += h;
408 p_minus[d] -= h;
409
410 this->vector_value(p_plus, vals_plus);
411 this->vector_value(p_minus, vals_minus);
412
413 // Centered finite differences
414 for (unsigned int c = 0; c < gradients.size(); ++c)
415 gradients[c][d] = (vals_plus[c] - vals_minus[c]) / (2.0 * h);
416 }
417 }
418
419 protected:
420 const ComponentOrdering &ordering;
421 const Parameters::PhysicalProperties<dim> &physical_properties;
422 ManufacturedSolutions::ManufacturedSolution<dim> mms;
