350 /**
351 * Gradient of source term, using finite differences
352 */
353 virtual void
354 vector_gradient(const Point<dim> &p,
355 std::vector<Tensor<1, dim>> &gradients) const override
356 {
357 const double h = 1e-8;
358
359 Vector<double> vals_plus(gradients.size()), vals_minus(gradients.size());
360
361 for (unsigned int d = 0; d < dim; ++d)
362 {
363 Point<dim> p_plus = p, p_minus = p;
364 p_plus[d] += h;
365 p_minus[d] -= h;
366
367 this->vector_value(p_plus, vals_plus);
368 this->vector_value(p_minus, vals_minus);
369
370 // Centered finite differences
371 for (unsigned int c = 0; c < gradients.size(); ++c)
372 gradients[c][d] = (vals_plus[c] - vals_minus[c]) / (2.0 * h);
373 }
374 }
