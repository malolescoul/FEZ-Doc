128template <int dim, bool with_moving_mesh>
129void CHNSSolver<dim, with_moving_mesh>::MMSSourceTerm::vector_value(
130 const Point<dim> &p,
131 Vector<double> &values) const
132{
133 const double phi = mms.exact_tracer->value(p);
134 const double filtered_phi = phi;
135 const double rho0 = physical_properties.fluids[0].density;
136 const double rho1 = physical_properties.fluids[1].density;
137 const double rho = CahnHilliard::linear_mixing(filtered_phi, rho0, rho1);
138 const double eta0 = rho0 * physical_properties.fluids[0].kinematic_viscosity;
139 const double eta1 = rho1 * physical_properties.fluids[1].kinematic_viscosity;
140 const double eta = CahnHilliard::linear_mixing(filtered_phi, eta0, eta1);
141 const double M = cahn_hilliard_param.mobility;
142 const double diff_flux_factor = M * 0.5 * (rho1 - rho0);
143 // const double drhodphi =
144 // CahnHilliard::linear_mixing_derivative(filtered_phi, rho0, rho1);
145 const double detadphi =
146 CahnHilliard::linear_mixing_derivative(filtered_phi, eta0, eta1);
147 const double epsilon = cahn_hilliard_param.epsilon_interface;
148 const double sigma_tilde =
149 3. / (2. * sqrt(2.)) * cahn_hilliard_param.surface_tension;
150 const double sigma_tilde_over_eps = sigma_tilde / epsilon;
151 const double sigma_tilde_times_eps = sigma_tilde * epsilon;
152 const auto &body_force = physical_properties.body_force;
153
154 Tensor<1, dim> u, dudt_eulerian;
155 for (unsigned int d = 0; d < dim; ++d)
156 {
157 dudt_eulerian[d] = mms.exact_velocity->time_derivative(p, d);
158 u[d] = mms.exact_velocity->value(p, d);
159 }
160
161 // Use convention (grad_u)_ij := dvj/dxi
162 Tensor<2, dim> grad_u = mms.exact_velocity->gradient_vj_xi(p);
163 Tensor<1, dim> lap_u = mms.exact_velocity->vector_laplacian(p);
164 Tensor<1, dim> grad_div_u = mms.exact_velocity->grad_div(p);
165 Tensor<1, dim> grad_p = mms.exact_pressure->gradient(p);
166 Tensor<1, dim> uDotGradu = u * grad_u;
167 Tensor<1, dim> grad_mu = mms.exact_potential->gradient(p);
168 Tensor<1, dim> grad_phi = mms.exact_tracer->gradient(p);
169 Tensor<1, dim> J_flux = diff_flux_factor * grad_mu;
170 Tensor<1, dim> div_viscous = (eta * (lap_u + grad_div_u) +
171 2. * detadphi * grad_phi * symmetrize(grad_u));
172
173 // Navier-Stokes momentum (velocity) source term
174 Tensor<1, dim> f = -(rho * (dudt_eulerian + uDotGradu - body_force) +
175 J_flux * grad_u + grad_p - div_viscous + phi * grad_mu);
176 for (unsigned int d = 0; d < dim; ++d)
177 values[u_lower + d] = f[d];
178
179 // Mass conservation (pressure) source term,
180 // for - div(u) + f = 0 -> f = div(u_mms).
181 values[p_lower] = mms.exact_velocity->divergence(p);
182
183 if constexpr (with_moving_mesh)
184 {
185 // Pseudosolid (mesh position) source term
186 Tensor<1, dim> f_PS =
187 mms.exact_mesh_position->divergence_elastic_stress_tensor(
188 physical_properties.pseudosolids[0], p);
189
190 for (unsigned int d = 0; d < dim; ++d)
191 values[x_lower + d] = f_PS[d];
192 }
193
194 // Tracer source term
195 const double dphidt = mms.exact_tracer->time_derivative(p);
196 const double lap_mu = mms.exact_potential->laplacian(p);
197 values[phi_lower] = -(dphidt + u * grad_phi - M * lap_mu);
198
199 // Potential source term
200 const double mu = mms.exact_potential->value(p);
201 const double lap_phi = mms.exact_tracer->laplacian(p);
202 values[mu_lower] = -(mu - sigma_tilde_over_eps * phi * (phi * phi - 1.) +
203 sigma_tilde_times_eps * lap_phi);
204}

206template <int dim, bool with_moving_mesh>
207void CHNSSolver<dim, with_moving_mesh>::update_simulation_parameters(
208 const unsigned int fixed_point_iteration)
209{
210 // Don't update anything at the first iteration.
211 if (fixed_point_iteration == 0)
212 return;
213
214 const auto &fpu = this->param.mesh.adaptation.metric.fixed_point_updates;
215
216 // Update the interface thickness
217 {
218 const auto &eps_update_data = fpu.chns_interface_thickness;
219 if (eps_update_data.enable &&
220 fixed_point_iteration % eps_update_data.update_frequency == 0)
221 {
222 auto &ch = this->param.cahn_hilliard;
223
224 if (fpu.verbosity == Parameters::Verbosity::verbose)
225 this->pcout << "-- Updating interface thickness from "
226 << ch.epsilon_interface << " to "
227 << ch.epsilon_interface * eps_update_data.factor
228 << std::endl;
229
230 // Update Cahn-Hilliard parameters
231 ch.epsilon_interface *= eps_update_data.factor;
232
233 // FIXME: Update mobility accordingly?
234
235 // Update initial condition
236 {
237 std::map<std::string, double> new_constants;
238 new_constants[eps_update_data.constant_name] = ch.epsilon_interface;
239 this->param.initial_conditions.initial_chns_tracer->update_constants(
240 new_constants);
241 }
242 }
243 }

363template <int dim, bool with_moving_mesh>
364void CHNSSolver<dim, with_moving_mesh>::create_sparsity_pattern()
365{
366 DynamicSparsityPattern dsp(this->locally_relevant_dofs);
367
368 const unsigned int n_components = this->ordering->n_components;
369 auto &coupling_table = this->coupling_table;
370 coupling_table = Table<2, DoFTools::Coupling>(n_components, n_components);
371 for (unsigned int i = 0; i < n_components; ++i)
372 for (unsigned int j = 0; j < n_components; ++j)
373 {
374 coupling_table[i][j] = DoFTools::none;
375
376 // u couples to all variables
377 if (this->ordering->is_velocity(i))
378 coupling_table[i][j] = DoFTools::always;
379
380 // p couples to u and x. PSPG also couples p to p, phi and mu through
381 // the strong momentum residual.
382 if (this->ordering->is_pressure(i) &&
383 (this->ordering->is_velocity(j) || this->ordering->is_position(j) ||
384 this->param.stabilization.enable_supg))
385 coupling_table[i][j] = DoFTools::always;
386
387 // x couples x,phi,u
388 if constexpr (with_moving_mesh)
389 if (this->ordering->is_position(i) &&
390 (this->ordering->is_position(j) || this->ordering->is_tracer(j) ||
391 this->ordering->is_velocity(j)))
392 coupling_table[i][j] = DoFTools::always;
393
394 // phi couples to u, phi, mu, x
395 if (this->ordering->is_tracer(i))
396 if (!this->ordering->is_pressure(j))
397 coupling_table[i][j] = DoFTools::always;
398
399 // mu couples to phi, mu, u, x
400 if (this->ordering->is_potential(i))
401 if (!this->ordering->is_pressure(j))
402 coupling_table[i][j] = DoFTools::always;
403 }
404
405 DoFTools::make_sparsity_pattern(*this->dof_handler,
406 coupling_table,
407 dsp,
408 this->nonzero_constraints,
409 /* keep_constrained_dofs = */ false);
410 SparsityTools::distribute_sparsity_pattern(dsp,
411 this->locally_owned_dofs,
412 this->mpi_communicator,
413 this->locally_relevant_dofs);
414 this->system_matrix.reinit(this->locally_owned_dofs,
415 this->locally_owned_dofs,
416 dsp,
417 this->mpi_communicator);
418}
